FORM get_carrid .

  CLEAR gs_input.

  SELECT SINGLE *
    INTO CORRESPONDING FIELDS OF gs_input
    FROM sflight
   WHERE carrid EQ pa_carr
     AND connid IN so_conn.

ENDFORM.
