MODULE exit_0100 INPUT.

ENDMODULE.

*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'SEARCH'.
      CLEAR gv_okcode.

      PERFORM get_carrid.

    WHEN 'GR'.
      CLEAR gv_okcode.

      MESSAGE i000 WITH 'GR 버튼 클릭'.

  ENDCASE.

ENDMODULE.
