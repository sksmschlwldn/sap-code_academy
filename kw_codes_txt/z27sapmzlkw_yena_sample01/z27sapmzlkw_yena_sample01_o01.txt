MODULE active_tab OUTPUT.

  gc_tab-activetab = gv_pressed.

  CASE gv_pressed.
    WHEN 'SCR01'.
      gv_screen = '0101'.

    WHEN 'SCR02'.
      gv_screen = '0102'.

  ENDCASE.

ENDMODULE.

*&---------------------------------------------------------------------*
*& Module SET_LAYOUT OUTPUT
*&---------------------------------------------------------------------*
*&
*&---------------------------------------------------------------------*
MODULE set_fcat_layout OUTPUT.

  IF gt_fcat IS INITIAL.

    PERFORM get_data.
    PERFORM set_fcat_layout.

  ENDIF.

ENDMODULE.
*&---------------------------------------------------------------------*
*& Form set_layout
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_fcat_layout .

  gs_layout = VALUE #( zebra      = 'X'
                       cwidth_opt = 'A'
                       sel_mode   = 'D' ).

  PERFORM set_fcat USING :
  'X'   'CARRID'      ' '   'SFLIGHT'   'CARRID',
  'X'   'CONNID'      ' '   'SFLIGHT'   'CONNID',
  'X'   'FLDATE'      ' '   'SFLIGHT'   'FLDATE',
  ' '   'PRICE'       ' '   'SFLIGHT'   'PRICE',
  ' '   'CURRENCY'    ' '   'SFLIGHT'   'CURRENCY',
  ' '   'PLANETYPE'   ' '   'SFLIGHT'   'PLANETYPE',
  ' '   'PAYMENTSUM'  ' '   'SFLIGHT'   'PAYMENTSUM'.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_data .

  REFRESH : gt_data.

  SELECT carrid    connid     fldate price currency
         planetype paymentsum
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM sflight
   WHERE carrid IN so_carr.

ENDFORM.
