MODULE get_pressed INPUT.

  gv_pressed = COND #( WHEN gv_okcode = 'SCR01' OR gv_okcode = 'SCR02'
                         THEN gv_okcode
                       ELSE gv_pressed ).

ENDMODULE.

MODULE user_command_0101 INPUT.

  CASE gv_okcode.
    WHEN 'SEARCH'.
      CLEAR gv_okcode.
      PERFORM get_data.
      PERFORM refresh_grid_1.

  ENDCASE.

ENDMODULE.
