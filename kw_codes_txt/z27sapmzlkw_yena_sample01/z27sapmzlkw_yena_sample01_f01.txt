FORM set_fcat  USING pv_key
                     pv_field
                     pv_text
                     pv_ref_table
                     pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'PRICE' OR 'PAYMENTSUM'.
      gs_fcat-cfieldname = 'CURRENCY'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen_1 .

  gcl_container_1 = NEW cl_gui_custom_container( container_name = 'GCL_CONTAINER_1' ).

  gcl_splitter    = NEW cl_gui_splitter_container(
                           parent  = gcl_container_1
                           rows    = 1
                           columns = 2 ).

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_l.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 2
    RECEIVING
      container = gcl_container_r.

  CALL METHOD gcl_splitter->set_column_width
    EXPORTING
      id    = 1
      width = 18.

  gcl_grid_1 = NEW cl_gui_alv_grid( i_parent = gcl_container_r ).

  gcl_calendar_1  = NEW cl_gui_calendar(
                          parent     = gcl_container_l
                          view_style = cnca_style_v_navigator + cnca_style_dtpicker ).

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid_1->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
