*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'UPLOAD'.
      CLEAR gv_okcode.

      PERFORM pdf_display.

    WHEN 'COLLAPSE'.

      PERFORM collapse_con.

    WHEN 'EXPAND'.

      PERFORM expand_con.

  ENDCASE.

ENDMODULE.
