FORM get_data .

  CLEAR   : gs_scarr, gs_spfli, gs_sflight.
  REFRESH : gt_scarr, gt_spfli, gt_sflight.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr
   WHERE carrid IN so_carr.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH 'Data not found'.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM create_object .

  IF gcl_container IS NOT BOUND.

    CREATE OBJECT gcl_container
      EXPORTING
        side      = gcl_container->dock_at_left
        extension = 5000.

    CREATE OBJECT gcl_splitter
      EXPORTING
        parent  = gcl_container
        rows    = 3
        columns = 1.

    CALL METHOD gcl_splitter->get_container
      EXPORTING
        row       = 1
        column    = 1
      RECEIVING
        container = gcl_container_top.

    CALL METHOD gcl_splitter->get_container
      EXPORTING
        row       = 2
        column    = 1
      RECEIVING
        container = gcl_container_mid.

    CALL METHOD gcl_splitter->get_container
      EXPORTING
        row       = 3
        column    = 1
      RECEIVING
        container = gcl_container_bot.

    CREATE OBJECT gcl_grid_top
      EXPORTING
        i_parent = gcl_container_top.

    CREATE OBJECT gcl_grid_mid
      EXPORTING
        i_parent = gcl_container_mid.

    CREATE OBJECT gcl_grid_bot
      EXPORTING
        i_parent = gcl_container_bot.

    IF gcl_handler IS NOT BOUND.
      CREATE OBJECT gcl_handler.
    ENDIF.

    SET HANDLER : gcl_handler->handle_double_click FOR gcl_grid_top,
                  gcl_handler->handle_hotspot      FOR gcl_grid_mid.

    gs_variant-report = sy-repid.
    gs_variant-handle = 'TP'.

    CALL METHOD gcl_grid_top->set_table_for_first_display
      EXPORTING
        i_save          = 'A'
        i_default       = 'X'
        is_layout       = gs_layout
        is_variant      = gs_variant
      CHANGING
        it_outtab       = gt_scarr
        it_fieldcatalog = gt_fcat_top.

*    CLEAR gs_layout-cwidth_opt.
    gs_variant-handle = 'MD'.

    CALL METHOD gcl_grid_mid->set_table_for_first_display
      EXPORTING
        i_save          = 'A'
        i_default       = 'X'
        is_layout       = gs_layout
        is_variant      = gs_variant
      CHANGING
        it_outtab       = gt_spfli
        it_fieldcatalog = gt_fcat_mid.

    gs_variant-handle = 'BO'.

    CALL METHOD gcl_grid_bot->set_table_for_first_display
      EXPORTING
        i_save          = 'A'
        i_default       = 'X'
        is_layout       = gs_layout
        is_variant      = gs_variant
      CHANGING
        it_outtab       = gt_sflight
        it_fieldcatalog = gt_fcat_bot.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_doubl_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click USING ps_row TYPE lvc_s_row
                                ps_col TYPE lvc_s_col.

  READ TABLE gt_scarr INTO gs_scarr INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'CARRID'.
      PERFORM get_schedule USING gs_scarr-carrid.
      PERFORM refresh_grid_mid.
      PERFORM refresh_grid_bot.

  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_schedule
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_SCARR_CARRID
*&---------------------------------------------------------------------*
FORM get_schedule USING pv_carrid.

  CLEAR   : gs_spfli, gs_sflight.
  REFRESH : gt_spfli, gt_sflight.

  SELECT carrid connid countryfr cityfrom countryto cityto
         fltime fltype
    INTO CORRESPONDING FIELDS OF TABLE gt_spfli
    FROM spfli
   WHERE carrid = pv_carrid.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH 'Schedule not found'.
    EXIT.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
FORM handle_hotspot USING ps_row TYPE lvc_s_row
                           ps_col TYPE lvc_s_col.

  READ TABLE gt_spfli INTO gs_spfli INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'CONNID'.
      PERFORM get_sflight USING gs_spfli-carrid gs_spfli-connid.
      PERFORM refresh_grid_bot.

  ENDCASE.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form get_sflight
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_SPFLI_CARRID
*&      --> GS_SPFLI_CONNID
*&---------------------------------------------------------------------*
FORM get_sflight USING pv_carrid pv_connid.

  CLEAR   gs_sflight.
  REFRESH gt_sflight.

  SELECT carrid    connid     fldate price currency
         planetype paymentsum
    INTO CORRESPONDING FIELDS OF TABLE gt_sflight
    FROM sflight
   WHERE carrid = pv_carrid
     AND connid = pv_connid.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH 'Flight data not found'.
    EXIT.
  ENDIF.

ENDFORM.
