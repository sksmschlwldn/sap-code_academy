MODULE get_pressed INPUT.

  CASE gv_okcode.
    WHEN 'SCARR'.
      gv_pressedtab = gv_okcode.
      gv_active1    = 'X'.

      CLEAR : gv_okcode, gv_active2.

    WHEN 'SPFLI'.
      gv_pressedtab = gv_okcode.
      gv_active2    = 'X'.

      CLEAR : gv_okcode, gv_active1.

    WHEN 'T1'.

      CASE 'X'.
        WHEN gv_active1.
          gv_pressedtab = 'SCARR'.

        WHEN gv_active2.
          gv_pressedtab = 'SPFLI'.

      ENDCASE.

  ENDCASE.

ENDMODULE.
