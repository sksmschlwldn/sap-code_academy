FORM make_number CHANGING pv_invno TYPE zc5lkwt0001-inv_no.

  DATA : lv_num(7) TYPE n,
         lv_range  TYPE inri-nrrangenr VALUE '02',
         lv_obj    TYPE inri-object    VALUE 'ZC531_TEST'.

  CALL FUNCTION 'NUMBER_GET_NEXT'
    EXPORTING
      nr_range_nr = lv_range
      object      = lv_obj
    IMPORTING
      number      = lv_num.

  CONCATENATE 'INV' lv_num INTO pv_invno.

ENDFORM.

FORM make_data_by_batch .

  DATA : ls_data  TYPE zc5lkwt0001,
         lv_invno TYPE zc5lkwt0001-inv_no.

  CLEAR : ls_data, lv_invno.

  PERFORM make_number CHANGING lv_invno.

  IF lv_invno IS INITIAL.
    lv_invno = 'INV3000000'.
  ENDIF.

  ls_data = VALUE #(
                      inv_no = lv_invno
                      text   = | INVOICE { lv_invno } |
                   ).

  MODIFY zc5lkwt0001 FROM ls_data.

  IF sy-dbcnt > 0.
    COMMIT WORK AND WAIT.
    MESSAGE s000 WITH 'Data saved'.
  ELSE.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'Data save failed' DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.
