FORM pdf_down .

  DATA : lv_sheet TYPE i,
         lv_tabix TYPE sy-tabix,
         lv_err.

  IF gcl_objfile IS NOT BOUND.
    CREATE OBJECT gcl_objfile.
  ENDIF.

  IF gv_pfolder IS NOT INITIAL.
    gv_initialfolder = gv_pfolder.
  ELSE.
    gcl_objfile->get_temp_directory( CHANGING temp_dir = gv_initialfolder
                                     EXCEPTIONS cntl_error           = 1
                                                error_no_gui         = 2
                                                not_supported_by_gui = 3 ).
  ENDIF.

  gcl_objfile->directory_browse( EXPORTING initial_folder        = gv_initialfolder
                                 CHANGING  selected_folder       = gv_pickedfolder
                                 EXCEPTIONS cntl_error           = 1
                                            error_no_gui         = 2
                                            not_supported_by_gui = 3 ).

  IF sy-subrc = 0.
    gv_pfolder = gv_pickedfolder.
  ELSE.
    MESSAGE TEXT-m01 TYPE 'S' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  IF gv_pfolder IS INITIAL.
    EXIT.
  ENDIF.

  CALL METHOD gcl_objfile->free.
  FREE gcl_objfile.

  CLEAR : gs_excel, gv_temp_filename, gv_filename.

  gv_form = 'ZLKW_PDF_DOWN'.

  CONCATENATE '\운항정보_' sy-datum sy-uzeit '.XLS'
  INTO gv_filename.

  CONCATENATE '\운항정보_' sy-datum sy-uzeit '.PDF'
  INTO gv_filename_pdf.

  CONDENSE : gv_filename, gv_filename_pdf.

  CONCATENATE gv_pfolder gv_filename
  INTO gv_temp_filename.

  CONCATENATE gv_pfolder gv_filename_pdf
  INTO gv_temp_filename_pdf.

  PERFORM download_template USING gv_form gv_temp_filename.

  PERFORM open_excel_template.

  lv_sheet = 1.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    PERFORM fill_list USING lv_tabix lv_sheet gs_data.

  ENDLOOP.

  PERFORM make_pdf USING 1.

ENDFORM.

FORM open_excel_template .

  IF gs_excel IS INITIAL.
    CREATE OBJECT gs_excel 'EXCEL.APPLICATION'.
  ENDIF.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH sy-msgli DISPLAY LIKE 'E'.
  ENDIF.

  CALL METHOD OF gs_excel 'WORKBOOKS' = gs_workbook.
  SET PROPERTY OF gs_excel 'VISIBLE' = 0 .

  CALL METHOD OF gs_workbook 'OPEN' EXPORTING #1 = gv_temp_filename.

* Sheet에대한 설정을 할때 사용된다.
  GET PROPERTY OF gs_workbook    'Application' = gs_application .
  GET PROPERTY OF gs_application 'ActiveSheet' = gs_activesheet.

ENDFORM.

FORM fill_list USING pv_line TYPE i
                      pv_sheet TYPE i
                      ps_data  LIKE gs_data.

  DATA : lv_line     TYPE i,
         lv_val(100).

* 엑셀에 1번째는 컬럼헤더 이므로 2번째 라인부터 찍어야 한다
  lv_line = pv_line + 1.

  PERFORM fill_cells USING lv_line 1 ps_data-carrid.
  PERFORM fill_cells USING lv_line 2 ps_data-connid.

* 날짜의 output 형식으로 맞추기
  CLEAR lv_val.
  WRITE ps_data-fldate TO lv_val.
  CONDENSE lv_val.

  PERFORM fill_cells USING lv_line 3 lv_val.
  PERFORM fill_cells USING lv_line 4 ps_data-planetype.

* 금액의 통화키 적용하여 output 형식으로 맞추기
  CLEAR lv_val.
  WRITE ps_data-price TO lv_val CURRENCY ps_data-currency.
  CONDENSE lv_val.

  PERFORM fill_cells USING lv_line 5 lv_val.
  PERFORM fill_cells USING lv_line 6 ps_data-currency.

ENDFORM.

      #2 = pv_j.

  SET PROPERTY OF gs_cell 'VALUE' = pv_val.

ENDFORM.

FORM make_pdf USING pv_sheet.

  DATA : lv_rc TYPE i.

  CALL METHOD OF gs_excel 'SHEETS' = gs_sheet
    EXPORTING #1 = 1.

  CALL METHOD OF gs_sheet 'SELECT' NO FLUSH.

* 모두 출력후 맨윗칸으로 커서 이동

  CALL METHOD OF gs_excel 'Cells' = gs_cell
    EXPORTING
    #1 = 1
    #2 = 1.

  CALL METHOD OF gs_cell 'Select'.

  SET PROPERTY OF gs_excel 'VISIBLE' = 1 . "엑셀 데이타를 다 뿌리고나서 보이기

** 만든 엑셀 지우고 PDF로 생성
  GET PROPERTY OF gs_excel 'Workbooks' = gs_workbook
    EXPORTING #1 = pv_sheet.

  CALL METHOD OF gs_workbook 'ExportAsFixedFormat'
    EXPORTING
      #1 = '0'
      #2 = gv_temp_filename_pdf.

  CALL METHOD OF gs_workbook 'Close'
    EXPORTING
      #1 = 0.

  CALL METHOD OF gs_excel 'Quit'.

  CALL METHOD cl_gui_frontend_services=>file_delete
    EXPORTING
      filename = CONV #( gv_temp_filename )
    CHANGING
      rc       = lv_rc.

  CALL METHOD OF gs_workbook 'ExportAsFixedFormat'
    EXPORTING
      #1 = 0
      #2 = gv_temp_filename_pdf.

ENDFORM.
