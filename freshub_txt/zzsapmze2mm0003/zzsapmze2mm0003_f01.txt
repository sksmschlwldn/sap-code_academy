*&---------------------------------------------------------------------*
*& Form make_hierarchy
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_hierarchy .

  DATA : lv_first,
         lv_text(12).

  DATA : BEGIN OF ls_mtart,
           mtart TYPE zte2mm0001-mtart,
         END OF ls_mtart,

         lt_mtart LIKE TABLE OF ls_mtart,

         BEGIN OF ls_matkl,
           mtart TYPE zte2mm0001-mtart,
           matkl TYPE zte2mm0001-matkl,
         END OF ls_matkl,

         lt_matkl    LIKE TABLE OF ls_matkl,
         lt_data     LIKE TABLE OF gs_mat,

         lv_level_1  TYPE lvc_nkey,
         lv_level_2  TYPE lvc_nkey,
         lv_node_key TYPE lvc_nkey,
         lv_tabix    TYPE sy-tabix,

* 검색 itab 구성키 위한 변수
         lv_key_1    TYPE tv_nodekey,
         lv_key_2    TYPE tv_nodekey,
         lv_cnt      TYPE sy-tabix,

* NODE 1,2 초기화
         ls_empty    LIKE gs_mat.

  CLEAR   : ls_mtart, ls_matkl, lv_first, lv_cnt, lv_key_1, lv_key_2, ls_empty.
  REFRESH : lt_data, lt_mtart, lt_matkl.

  APPEND LINES OF gt_mat TO lt_data.

* 트리구성시 gt_mat에 노드가 쌓이는 구조.
* 기존 데이터는 lt_data에 보관 후 gt_mat에 다시 쌓음.
  REFRESH : gt_mat.

  SORT lt_data BY mtart matkl matnr.

  lt_mtart = CORRESPONDING #( lt_data ).
  lt_matkl = CORRESPONDING #( lt_data ).

  SORT : lt_mtart BY mtart,
         lt_matkl BY mtart matkl.

  DELETE ADJACENT DUPLICATES FROM : lt_mtart COMPARING mtart,
                                    lt_matkl COMPARING mtart matkl.

  LOOP AT lt_mtart INTO ls_mtart.

    READ TABLE lt_mtart INTO ls_mtart WITH KEY mtart = ls_mtart-mtart
    BINARY SEARCH.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    lv_tabix = sy-tabix.

    lv_cnt  += 1.
    lv_key_1 = lv_cnt.

    READ TABLE lt_data INTO gs_mat WITH KEY mtart = ls_mtart-mtart
    BINARY SEARCH.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    CASE gs_mat-mtart.
      WHEN 'FERT'.
        lv_text = '완제품'.
      WHEN 'ROH'.
        lv_text = '구매재'.
      WHEN OTHERS.
        lv_text = '포장재'.
    ENDCASE.

* Level 1 Node
    PERFORM add_node USING ls_empty lv_text ' ' lv_level_1.

    LOOP AT lt_matkl INTO ls_matkl FROM lv_tabix WHERE mtart = ls_mtart-mtart.

      READ TABLE lt_data INTO gs_mat WITH KEY mtart = ls_mtart-mtart
                                              matkl = ls_matkl-matkl
                                              BINARY SEARCH.
      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      lv_tabix = sy-tabix.

      CLEAR lv_first.

      LOOP AT lt_data INTO gs_mat FROM lv_tabix WHERE mtart = ls_mtart-mtart
                                                  AND matkl = ls_matkl-matkl.

        IF lv_first IS INITIAL.

          CASE gs_mat-matkl.
            WHEN 'AG'.
              lv_text = '농산물'.
            WHEN 'NV'.
              lv_text = '축산물'.
            WHEN 'SF'.
              lv_text = '수산물'.
            WHEN 'PR'.
              lv_text = '드레싱'.
            WHEN 'PM'.
              lv_text = '신선식품'.
            WHEN 'IP'.
              lv_text = '내부 포장재'.
            WHEN 'OP'.
              lv_text = '외부 포장재'.
            WHEN 'PP'.
              lv_text = '생산용 포장재'.
            WHEN OTHERS.
          ENDCASE.

* Level 2 Node
          PERFORM add_node USING ls_empty lv_text lv_level_1 lv_level_2.
          lv_first ='X'.
          lv_cnt += 1.
          lv_key_2 = lv_cnt.

        ENDIF.

* Child Node
        lv_cnt += 1.
        PERFORM add_node USING gs_mat gs_mat-maktx lv_level_2 lv_node_key.

        gs_node_key = CORRESPONDING #( gs_mat ).
        gs_node_key = VALUE #( BASE gs_node_key
                               node_key_1 = lv_key_1
                               node_key_2 = lv_key_2 ).

        APPEND gs_node_key TO gt_node_key.
        CLEAR  gs_node_key.

      ENDLOOP.

    ENDLOOP.

  ENDLOOP.

  CALL METHOD gcl_tree->update_calculations.
  CALL METHOD gcl_tree->frontend_update.

  SORT gt_node_key BY node_key_1 node_key_2.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> NODE_KEY
*&---------------------------------------------------------------------*
FORM handle_double_click  USING  pv_node_key TYPE lvc_nkey.

  DATA : ls_outtab    LIKE gs_mat,
         lv_tax_txt   LIKE gs_info-tax_txt,
         lv_matkl_txt LIKE gs_info-matkl_txt,
         lt_domain    TYPE TABLE OF dd07v,
         ls_domain    LIKE LINE OF lt_domain.

  CALL METHOD gcl_tree->get_outtab_line
    EXPORTING
      i_node_key    = pv_node_key
    IMPORTING
      e_outtab_line = ls_outtab.

  IF ls_outtab-matnr IS INITIAL.
    EXIT.
  ENDIF.

  SELECT SINGLE
           a~matnr, a~maktx, a~mtart, a~matkl, a~dislf,  a~lvorm,
           a~meins, a~raube, a~mhdhb, a~iprkz, a~safstk, a~bstme,
           a~mwskz, a~ntgew, a~gewei,
           CASE
             WHEN b~total_stock IS NULL THEN 0
             WHEN b~total_stock < 0     THEN 0 " 혹시 마이너스 재고면 0 처리
             ELSE b~total_stock
           END AS total_stock,
           a~meins AS stock_meins,
           CASE
             WHEN c~stock_status = '정상' THEN '@5B@'
             WHEN c~stock_status = '과잉'
               OR c~stock_status = '위험'
               OR c~stock_status = '확인필요'  THEN '@5C@'
             WHEN c~stock_status = '주의' THEN '@5D@'
           ELSE '@5F@'
           END AS stat_icon,
           c~stock_status AS stat_txt

      FROM zte2mm0001 AS a
      LEFT OUTER JOIN zcds_e2_pp_0001 AS b
        ON a~matnr = b~matnr
     INNER JOIN zcds_e2_mm_0002 AS c
        ON a~matnr = c~matnr
     WHERE a~matnr = @ls_outtab-matnr
      INTO CORRESPONDING FIELDS OF @gs_info.

  " 세금 내역
  SELECT SINGLE text1
    FROM zte2fi0010
   WHERE mwskz = @gs_info-mwskz
    INTO @lv_tax_txt.

  gs_info-tax_txt = lv_tax_txt.

  " 품목분류 내역
  CALL FUNCTION 'GET_DOMAIN_VALUES'
    EXPORTING
      domname    = 'ZDE2_MM_MATKL'
      text       = 'X'
    TABLES
      values_tab = lt_domain.

  READ TABLE lt_domain INTO ls_domain
    WITH KEY domvalue_l = gs_info-matkl.

  IF sy-subrc = 0.
    gs_info-matkl_txt = ls_domain-ddtext.
  ENDIF.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  PERFORM get_mat_data USING ls_outtab-matnr
                             ls_outtab-mtart.

  CALL METHOD cl_gui_cfw=>set_new_ok_code
    EXPORTING
      new_code = 'GO_PAI'.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_mat_data USING pv_matnr TYPE zte2mm0001-matnr
                        pv_mtart TYPE zte2mm0001-mtart.


  CASE pv_mtart.

    WHEN 'FERT'.
      REFRESH gt_bom.

      SELECT a~idnrk, b~maktx, a~menge, a~meins
        INTO CORRESPONDING FIELDS OF TABLE @gt_bom
        FROM zte2pp0003 AS a
       INNER JOIN zte2mm0001 AS b ON a~idnrk = b~matnr
       INNER JOIN zte2pp0002 AS c ON a~stlnr = c~stlnr
       WHERE c~matnr = @pv_matnr.

      IF sy-subrc <> 0.
        MESSAGE s000 WITH '자재 공급원 정보가 존재하지 않습니다.' DISPLAY LIKE 'W'.

      ENDIF.

    WHEN OTHERS.
      REFRESH gt_partner.

      SELECT a~partner, b~company, a~minbm, a~meins, a~netpr, a~waers, a~plifz
        FROM zte2mm0006 AS a
       INNER JOIN zte2us0001 AS b
          ON a~partner = b~partner
       WHERE matnr = @pv_matnr
        INTO CORRESPONDING FIELDS OF TABLE @gt_partner.

      IF sy-subrc <> 0.
        MESSAGE s000 WITH '자재 공급원 정보가 존재하지 않습니다.' DISPLAY LIKE 'W'.

      ENDIF.

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM expand_tree .

  DATA : lv_node_key TYPE tv_nodekey.

  IF gs_search-matnr IS NOT INITIAL.

    " 트리를 펼치기 위한 노드 키 세팅.
    READ TABLE gt_node_key INTO gs_node_key WITH KEY matnr = gs_search-matnr.
    lv_node_key = gs_node_key-node_key_2.

  ELSEIF gs_search-matkl IS NOT INITIAL.

    READ TABLE gt_node_key INTO gs_node_key WITH KEY matkl = gs_search-matkl.

    lv_node_key = gs_node_key-node_key_2.

  ELSE.
    READ TABLE gt_node_key INTO gs_node_key WITH KEY mtart = gs_search-mtart.

    lv_node_key = gs_node_key-node_key_1.

  ENDIF.

  gcl_tree->collapse_all_nodes( ). " 이전에 펼쳐진 모든 트리를 닫음.

  gcl_tree->expand_node(
    EXPORTING
      i_node_key       = lv_node_key
      i_expand_subtree = 'X'
    EXCEPTIONS
      failed              = 1
      illegal_level_count = 2
      cntl_system_error   = 3
      node_not_found      = 4
      cannot_expand_leaf  = 5
      OTHERS              = 6 ).

  IF sy-subrc NE 0.
    MESSAGE s000 WITH '트리 펼치기에 실패했습니다.' DISPLAY LIKE 'E'.
  ENDIF.


ENDFORM.

*&---------------------------------------------------------------------*
*& Form get_f4_matkl
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_f4_matkl USING  pv_mtart pv_matkl.

  DATA : BEGIN OF ls_matkl,
           matkl TYPE zte2mm0001-matkl,
           desc  TYPE zee2_mm_matkl_txt,
         END OF ls_matkl,

         lt_matkl  LIKE TABLE OF ls_matkl,

         lt_field  TYPE TABLE OF dynpread,
         ls_field  LIKE LINE OF lt_field,
         lv_subrc  TYPE sy-subrc,

         lt_domain TYPE TABLE OF dd07v,
         ls_domain LIKE LINE OF lt_domain,

         lv_flag.

  CLEAR : ls_matkl.

  ls_field-fieldname = pv_mtart.

  APPEND ls_field TO lt_field.

  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname               = sy-repid  " 현재 프로그램명
      dynumb               = sy-dynnr  " 현재 화면번호
    TABLES
      dynpfields           = lt_field  " 읽어올 필드 목록
    EXCEPTIONS
      invalid_abapworkarea = 1
      invalid_dynprofield  = 2
      invalid_dynproname   = 3
      invalid_dynpronummer = 4.

  " 읽어온 화면에서 자재유형 꺼내오기
  READ TABLE lt_field INTO ls_field WITH KEY fieldname = pv_mtart.

  CALL FUNCTION 'GET_DOMAIN_VALUES'
    EXPORTING
      domname    = 'ZDE2_MM_MATKL'
      text       = 'X'
    TABLES
      values_tab = lt_domain.

  " 보여줄 도메인 값만 세팅하기 ..
  LOOP AT lt_domain INTO ls_domain.

    CLEAR lv_flag.

    CASE ls_field-fieldvalue.
      WHEN 'FERT'.
        IF ls_domain-domvalue_l = 'PM'.
          lv_flag = 'X'.

FORM validate_search_data CHANGING pv_flag.

  DATA  : lv_err.
  CLEAR : lv_err, pv_flag.

  IF gs_search-mtart IS NOT INITIAL AND gs_search-matkl IS NOT INITIAL.
    CASE gs_search-mtart.
      WHEN 'FERT'.
        IF gs_search-matkl <> 'PM'.
          lv_err = 'X'.
        ENDIF.

        IF gs_search-matkl <> 'IP' AND gs_search-matkl <> 'OP' AND gs_search-matkl <> 'PP'.
          lv_err = 'X'.
        ENDIF.
      WHEN 'ROH'.

           gs_search-matkl <> 'SF' AND gs_search-matkl <> 'PR'.
          lv_err = 'X'.
        ENDIF.
    ENDCASE.

    IF lv_err IS NOT INITIAL.
      MESSAGE s000 WITH '해당 자재유형에 없는 품목분류입니다.' DISPLAY LIKE 'E'.
      pv_flag = 'X'.
      RETURN.
    ENDIF.
  ENDIF.

  IF gs_search-mtart IS NOT INITIAL AND gs_search-matnr IS NOT INITIAL.
    SELECT SINGLE matnr
      FROM zte2mm0001
     WHERE mtart = @gs_search-mtart
       AND matnr = @gs_search-matnr
      INTO @DATA(lv_matnr).

    IF lv_matnr IS INITIAL.
      MESSAGE s000 WITH '해당 자재유형에 없는 자재입니다.' DISPLAY LIKE 'E'.
      pv_flag = 'X'.
      RETURN.
    ENDIF.
  ENDIF.

  IF gs_search-matkl IS NOT INITIAL AND gs_search-matnr IS NOT INITIAL.
    SELECT SINGLE matnr
      FROM zte2mm0001
     WHERE mtart = @gs_search-matkl
       AND matnr = @gs_search-matnr
      INTO @DATA(lv_matnr2).

    IF lv_matnr2 IS INITIAL.
      MESSAGE s000 WITH '해당 품목분류에 없는 자재입니다.' DISPLAY LIKE 'E'.
      pv_flag = 'X'.
      RETURN.
    ENDIF.
  ENDIF.

FORM set_info .

  SELECT SINGLE
             a~matnr, a~maktx, a~mtart, a~matkl, a~dislf,  a~lvorm,
             a~meins, a~raube, a~mhdhb, a~iprkz, a~safstk, a~bstme,
             CASE
               WHEN b~total_stock IS NULL THEN 0
               WHEN b~total_stock < 0     THEN 0 " 혹시 마이너스 재고면 0 처리
               ELSE b~total_stock
             END AS total_stock,
             a~meins AS stock_meins,
             CASE
               WHEN c~stock_status = '정상' THEN '@5B@'
               WHEN c~stock_status = '과잉'
                 OR c~stock_status = '위험'
                 OR c~stock_status = '확인필요'  THEN '@5C@'
               WHEN c~stock_status = '주의' THEN '@5D@'
             ELSE '@5F@'
             END AS stat_icon,
             c~stock_status AS stat_txt

        FROM zte2mm0001 AS a
        LEFT OUTER JOIN zcds_e2_pp_0001 AS b
          ON a~matnr = b~matnr
       INNER JOIN zcds_e2_mm_0002 AS c
          ON a~matnr = c~matnr
       WHERE a~matnr = @gs_search-matnr
        INTO CORRESPONDING FIELDS OF @gs_info.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  PERFORM get_mat_data USING gs_search-matnr
                             gs_info-mtart.

  CALL METHOD cl_gui_cfw=>set_new_ok_code
    EXPORTING
      new_code = 'GO_PAI'.

ENDFORM.
