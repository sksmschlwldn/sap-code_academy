MODULE exit_0100 INPUT.

  IF gcl_grid_mat IS NOT INITIAL.

    CALL METHOD : gcl_grid_mat->free, gcl_container_mat->free.
    FREE : gcl_grid_mat, gcl_container_mat.

MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'SEARCH'.
      CLEAR gv_okcode.
      PERFORM refresh_screen.

      DATA : lv_flag.

      IF gs_search IS NOT INITIAL.
        PERFORM validate_search_data CHANGING lv_flag .
        CHECK lv_flag IS INITIAL. " 에러있으면 트리펼치기 x
        PERFORM expand_tree. " 검색시 해당 검색조건에 맞는 트리의 노드를 펼쳐줌
        PERFORM set_info.
      ENDIF.

    WHEN 'REFRESH'.
      CLEAR gv_okcode.
      CLEAR   gs_search.
      PERFORM refresh_screen.

    WHEN 'CREATE'.
      CLEAR gv_okcode.
      CLEAR gs_info.

      PERFORM create_new_material.

    WHEN 'EDIT'.
      CLEAR gv_okcode.
      gv_mode = 'E'.

    WHEN 'SAVE'.
      CLEAR gv_okcode.
      DATA lv_flag3.

      PERFORM valid_fields CHANGING lv_flag3.
      CHECK lv_flag3 IS INITIAL.
      PERFORM update_material.
*       gv_mode = 'D'.

  ENDCASE.

ENDMODULE.

MODULE set_maktx INPUT.

  IF gs_search-matnr IS INITIAL.
    CLEAR gs_search-maktx.
    RETURN.
  ENDIF.

  SELECT SINGLE maktx
    INTO gs_search-maktx
    FROM zte2mm0001
   WHERE matnr = gs_search-matnr.

ENDMODULE.
