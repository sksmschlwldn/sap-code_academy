MODULE exit_0100 INPUT.

  CASE gv_okcode.
    WHEN 'BACK' OR 'EXIT' OR 'CANC'.
      CALL METHOD : gcl_grid_i->free, gcl_grid_h->free,
                  gcl_container_i->free, gcl_container_h->free.

      FREE: gcl_grid_i, gcl_grid_h, gcl_container_i, gcl_container_h.

      LEAVE TO SCREEN 0.
  ENDCASE.

ENDMODULE.

MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'REFRESH'.
      CLEAR gv_okcode.
      CLEAR gv_sel_bom.

      PERFORM get_bom_h.
      PERFORM get_bom_i USING gv_sel_bom.
      PERFORM set_style.
      PERFORM refresh_grid_h.
      PERFORM refresh_grid_i.

    WHEN 'SAVE' OR 'SAVE_BOM'.
      CLEAR gv_okcode.
      gv_mode = 'D'.
      PERFORM modify_screen.

      PERFORM save_data.
      PERFORM get_bom_h.
      PERFORM get_bom_i USING gv_sel_bom.
      PERFORM set_style.
      PERFORM refresh_grid_h.
      PERFORM refresh_grid_i.

    WHEN 'GO_PAI'.
      RETURN.

  ENDCASE.

ENDMODULE.
