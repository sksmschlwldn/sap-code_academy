*&---------------------------------------------------------------------*
*& Form get_bom_h
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_bom_h .

  DATA : lr_maktx TYPE RANGE OF zte2mm0001-maktx,
         ls_maktx LIKE LINE OF lr_maktx.

  CLEAR : gs_bom_h, ls_maktx.
  REFRESH : gt_bom_h, lr_maktx.

  IF pa_maktx IS NOT INITIAL.

    ls_maktx-sign = 'I'.
    ls_maktx-low = '*' && pa_maktx && '*'.
    ls_maktx-option = 'CP'.

    APPEND ls_maktx TO lr_maktx.

  ENDIF.

  SELECT a~stlnr, a~matnr, b~maktx, a~werks, a~bmeng, a~meins,
         CASE a~status
         WHEN 'A' THEN '활성'
         WHEN 'I' THEN '비활성'
         END AS stat_txt,
         a~status
    FROM zte2pp0002 AS a
   INNER JOIN zte2mm0001 AS b
      ON a~matnr = b~matnr
   WHERE a~matnr IN @so_matnr
     AND a~werks = @pa_werks
     AND b~maktx IN @lr_maktx
     AND a~stlnr IN @so_stlnr
   ORDER BY a~stlnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_bom_h.


ENDFORM.

FORM handle_hotspot  USING   ps_row TYPE lvc_s_row
                            ps_col TYPE lvc_s_col.

  gv_mode = 'E'.
  PERFORM modify_screen.

  READ TABLE gt_bom_h INTO gs_bom_h INDEX ps_row-index.

  IF sy-subrc <> 0.

  ENDIF.

  gv_sel_bom = gs_bom_h-stlnr.

    WHEN 'STLNR'.
      PERFORM get_bom_i USING gs_bom_h-stlnr.
      PERFORM set_style.
      PERFORM refresh_grid_i.

      CALL METHOD cl_gui_cfw=>set_new_ok_code
        EXPORTING

  ENDCASE.

*&---------------------------------------------------------------------*
FORM get_bom_i  USING pv_stlnr TYPE zte2pp0002-stlnr.

  CLEAR   : gs_bom_i.
  REFRESH : gt_bom_i.

  SELECT a~stlnr a~posnr a~idnrk b~maktx a~menge a~meins
    INTO CORRESPONDING FIELDS OF TABLE gt_bom_i
    FROM zte2pp0003 AS a
   INNER JOIN zte2mm0001 AS b
      ON a~idnrk = b~matnr
   WHERE a~stlnr = pv_stlnr
   ORDER BY a~posnr.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form delete_bom_h
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM delete_bom_h .

  DATA : lv_cnt    TYPE i,
         ls_del    LIKE LINE OF gt_bom_h_del,
         lv_answer.

  CALL METHOD gcl_grid_h->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  lv_cnt = lines( gt_row ).

  IF lv_cnt < 1.
    MESSAGE s000 WITH '먼저 행을 선택해주세요.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  " 진짜 삭제하시겠습니까?? 팝업.
  "  이때 해당 BOM이 사용되고있는 곳을 보여줘야함. EX. 생산 오더 등 ...

  CALL FUNCTION 'POPUP_TO_CONFIRM'
    EXPORTING
      titlebar              = 'BOM 전체 삭제'         " 팝업창 제목
      text_question         = 'BOM을 전체 삭제하시겠습니까?' " 물어볼 내용
      text_button_1         = '예'(001)           " 버튼 1 텍스트
      text_button_2         = '아니오'(002)       " 버튼 2 텍스트
      display_cancel_button = ' '                 " 취소 버튼은 필요 없으니 X
    IMPORTING
      answer                = lv_answer           " 사용자가 뭘 눌렀는지 담는 변수
    EXCEPTIONS
      text_not_found        = 1
      OTHERS                = 2.

  " 사용자가 '예'를 눌렀을 때만 삭제
  IF lv_answer = '1'.
    SORT  gt_row BY index DESCENDING.

    LOOP AT gt_row INTO gs_row.

      READ TABLE gt_bom_h INTO gs_bom_h INDEX gs_row-index.

      IF sy-subrc = 0.

        MOVE-CORRESPONDING gs_bom_h TO ls_del.

        APPEND ls_del TO gt_bom_h_del.
        CLEAR ls_del.

        DELETE gt_bom_h INDEX gs_row-index.
        REFRESH gt_bom_i.

      ENDIF.

    ENDLOOP.

  ELSE.
    MESSAGE '삭제가 취소되었습니다.' TYPE 'S' DISPLAY LIKE 'W'.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form delete_bom_i
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM delete_bom_i .

  DATA : lv_cnt TYPE i,
         ls_del LIKE LINE OF gt_bom_i_del.


  CALL METHOD gcl_grid_i->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  lv_cnt = lines( gt_row ).

  IF lv_cnt < 1.
    MESSAGE s000 WITH '먼저 행을 선택해주세요.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT  gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_bom_i INTO gs_bom_i INDEX gs_row-index.

    IF sy-subrc = 0.

      MOVE-CORRESPONDING gs_bom_i TO ls_del.
      gv_del_bom = gs_bom_i-stlnr.

      APPEND ls_del TO gt_bom_i_del.
      CLEAR ls_del.

      DELETE gt_bom_i INDEX gs_row-index.
      MESSAGE '삭제가 완료되었습니다.' TYPE 'S'.

    ENDIF.

  ENDLOOP.

ENDFORM.

         lt_save_h LIKE TABLE OF ls_save_h,
         lt_save_i LIKE TABLE OF ls_save_i,
         lv_commit.

  CLEAR : ls_save_h, ls_save_i, lv_commit.
  REFRESH : lt_save_h, lt_save_i.

  CALL METHOD gcl_grid_h->check_changed_data.

  " 헤더 삭제 시 삭제
*  IF gt_bom_h_del IS NOT INITIAL.
*
*    DELETE zte2pp0002 FROM TABLE gt_bom_h_del. " 헤더테이블 삭제
*    DELETE FROM zte2pp0003 WHERE stlnr = gv_sel_bom . " 헤더테이블 삭제
*
*    IF sy-dbcnt > 0.
*      lv_commit = 'X'.
*    ENDIF.
*
*  ENDIF.

  gs_bom_h = VALUE #( BASE gs_bom_h
                      aenam = sy-uname
                      aedat = sy-datum
                      aezet = sy-uzeit
  ).

  MOVE-CORRESPONDING gs_bom_h TO ls_save_h.

  MODIFY zte2pp0002 FROM ls_save_h.

  IF sy-dbcnt > 0.
    lv_commit = 'X'.
  ENDIF.

  " 아이템 삭제시 삭제
  IF gt_bom_i_del IS NOT INITIAL.

    DELETE zte2pp0003 FROM TABLE gt_bom_i_del.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  " 아이템 수정시 및 추가시 저장
  LOOP AT gt_bom_i INTO gs_bom_i WHERE mark = 'X'.

    MOVE-CORRESPONDING gs_bom_i TO ls_save_i.

    IF gs_bom_i-erdat IS INITIAL.
      ls_save_i-erdat = sy-datum.
      ls_save_i-erzet = sy-uzeit.
      ls_save_i-ernam = sy-uname.

    ELSE.
      ls_save_i-aedat = sy-datum.
      ls_save_i-aezet = sy-uzeit.
      ls_save_i-aenam = sy-uname.

    ENDIF.

    APPEND ls_save_i TO lt_save_i.
    CLEAR  ls_save_i.

  ENDLOOP.

  IF lt_save_i IS NOT INITIAL.

    MODIFY zte2pp0003 FROM TABLE lt_save_i.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.
    REFRESH gt_bom_i_del.
    PERFORM clear_mark.

    MESSAGE s000 WITH 'BOM이 성공적으로 변경되었습니다.'.

  ELSE.
    MESSAGE s000 WITH '변경된 BOM이 없습니다.' DISPLAY LIKE 'W'.
  ENDIF.

ENDFORM.

*& Form save_new_bom
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM save_new_bom .

  DATA : ls_save_h TYPE zte2pp0002,
         ls_save_i TYPE zte2pp0003,
         lt_save_h LIKE TABLE OF ls_save_h,
         lt_save_i LIKE TABLE OF ls_save_i,
         lv_commit,
         lv_exists TYPE abap_bool.

  CLEAR : ls_save_h, ls_save_i, lv_commit.
  REFRESH : lt_save_h, lt_save_i.

  CALL METHOD gcl_grid_h->check_changed_data.

  " 필수값 체크
  IF gs_bom_new_h-stlnr IS INITIAL.
    MESSAGE s000 WITH 'BOM 번호를 입력하세요.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  IF gs_bom_new_h-matnr IS INITIAL.
    MESSAGE s000 WITH '자재번호를 입력하세요.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  " 이미 존재하는 BOM 번호인지 체크
  SELECT SINGLE @abap_true
    FROM zte2pp0002
    WHERE stlnr = @gs_bom_new_h-stlnr
    INTO @lv_exists.

  IF lv_exists = abap_true.
    MESSAGE s000 WITH '이미 존재하는 BOM 번호입니다.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  " 헤더 저장
  MOVE-CORRESPONDING gs_bom_new_h TO ls_save_h.

  ls_save_h = VALUE #( BASE ls_save_h
                         ernam = sy-uname
                         erdat = sy-datum
                         erzet = sy-uzeit ).

  INSERT zte2pp0002 FROM ls_save_h.

  IF sy-subrc <> 0.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'BOM 헤더 저장 중 오류가 발생했습니다.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  " 아이템 저장
  LOOP AT gt_bom_i INTO gs_bom_i.

    IF gs_bom_i-idnrk IS INITIAL.
      CONTINUE.
    ENDIF.

    MOVE-CORRESPONDING gs_bom_i TO ls_save_i.

    ls_save_i-stlnr = gs_bom_new_h-stlnr.
    ls_save_i-erdat = sy-datum.
    ls_save_i-erzet = sy-uzeit.
    ls_save_i-ernam = sy-uname.

    APPEND ls_save_i TO lt_save_i.

  ENDLOOP.

  IF lt_save_i IS INITIAL.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'BOM 아이템을 입력하세요.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  INSERT zte2pp0003 FROM TABLE lt_save_i.

  IF sy-subrc <> 0.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'BOM 아이템 저장 중 오류가 발생했습니다.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.

  COMMIT WORK AND WAIT.

  MESSAGE s000 WITH 'BOM이 성공적으로 생성되었습니다.'.


ENDFORM.

*& Form f4_matnr

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM f4_matnr .

  DATA : BEGIN OF ls_matnr,
           matnr TYPE zte2pp0002-matnr,
           maktx TYPE zte2mm0001-maktx,
         END OF ls_matnr,

         lt_matnr  LIKE TABLE OF ls_matnr,

         lt_return TYPE TABLE OF ddshretval,
         ls_return TYPE ddshretval,
         lt_dynp   TYPE TABLE OF dynpread,
         ls_dynp   TYPE dynpread.

  SELECT matnr, maktx
    FROM zte2mm0001
   WHERE mtart = 'FERT'
    INTO CORRESPONDING FIELDS OF TABLE @lt_matnr.

CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'MATNR'
      dynpprog     = sy-repid
      dynpnr       = sy-dynnr
      dynprofield  = 'GS_BOM_NEW_H'
      window_title = '생산품 목록'
      value_org    = 'S'
    TABLES
      value_tab    = lt_matnr
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INTO ls_return INDEX 1.

    IF sy-subrc = 0.

      READ TABLE lt_matnr INTO ls_matnr WITH KEY matnr = ls_return-fieldval.

      IF sy-subrc = 0.
        CLEAR   ls_dynp.

        ls_dynp-fieldname = 'GS_BOM_NEW_H-MATNR'.
        ls_dynp-fieldvalue = ls_return-fieldval.
        APPEND ls_dynp TO lt_dynp.
        CLEAR ls_dynp.

        ls_dynp-fieldname = 'GS_BOM_NEW_H-MAKTX'.
        ls_dynp-fieldvalue = ls_matnr-maktx.
        APPEND ls_dynp TO lt_dynp.
        CLEAR ls_dynp.

        CALL FUNCTION 'DYNP_VALUES_UPDATE'
          EXPORTING
            dyname               = sy-repid
            dynumb               = sy-dynnr
          TABLES
            dynpfields           = lt_dynp
          EXCEPTIONS
            invalid_abapworkarea = 1
            invalid_dynprofield  = 2
            invalid_dynproname   = 3
            invalid_dynpronummer = 4
            invalid_request      = 5
            no_fielddescription  = 6
            undefind_error       = 7
            OTHERS               = 8.

      ENDIF.

      gs_bom_new_h-matnr = ls_matnr-matnr.
      gs_bom_new_h-maktx = ls_matnr-maktx.

    ENDIF.

  ENDIF.

ENDFORM.
