*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_order .

  DATA : lr_apstat TYPE RANGE OF zte2pp0009-apstat, " 승인 상태
         ls_apstat LIKE LINE OF lr_apstat,

         lr_gstrp  TYPE RANGE OF zte2pp0009-gstrp, " 날짜
         ls_gstrp  LIKE LINE OF lr_gstrp,

         lr_plnbez TYPE RANGE OF zte2pp0009-plnbez, " 자재 번호
         ls_plnbez LIKE LINE OF  lr_plnbez,

         lr_auart  TYPE RANGE OF zte2pp0009-auart,
         ls_auart  LIKE LINE OF lr_auart,

         lv_subrc  TYPE sy-subrc.

  REFRESH gt_order.

  " 승인상태 조건
  IF gs_search-apstat IS NOT INITIAL.
    ls_apstat-sign   = 'I'.
    ls_apstat-option = 'EQ'.
    ls_apstat-low    = gs_search-apstat.

    APPEND ls_apstat TO lr_apstat.
    CLEAR  ls_apstat.

  ENDIF.

  " 날짜 조건
  IF gs_search-gstrp_low IS NOT INITIAL.

    ls_gstrp-sign   = 'I'.
    ls_gstrp-low    = gs_search-gstrp_low.

    IF gs_search-gstrp_high IS INITIAL.
      ls_gstrp-option = 'GE'.
    ELSE.
      ls_gstrp-option = 'BT'.
      ls_gstrp-high   = gs_search-gstrp_high.
    ENDIF.

    APPEND ls_gstrp TO lr_gstrp.
    CLEAR  ls_gstrp.

  ENDIF.

  IF gs_search-matnr IS NOT INITIAL.

    PERFORM check_valid_matnr USING    gs_search-matnr
                              CHANGING lv_subrc.

    IF lv_subrc <> 0.
      EXIT.
    ENDIF.

    ls_plnbez-sign   = 'I'.
    ls_plnbez-option = 'EQ'.
    ls_plnbez-low    = gs_search-matnr.

    APPEND ls_plnbez TO lr_plnbez.
    CLEAR  ls_plnbez.

  ENDIF.

  " 오더구분 조건 (라디오버튼)
  CASE 'X'.
    WHEN gv_act_mto. "판매오더
      ls_auart-sign   = 'I'.
      ls_auart-option = 'EQ'.
      ls_auart-low    = 'ZSOR'.

      APPEND ls_auart TO lr_auart.
      CLEAR  ls_auart.

    WHEN gv_act_mts. "계획오더
      ls_auart-sign   = 'I'.
      ls_auart-option = 'EQ'.
      ls_auart-low    = 'ZFCT'.

      APPEND ls_auart TO lr_auart.
      CLEAR  ls_auart.

    WHEN gv_act_stk. " 부족 재고
      ls_auart-sign   = 'I'.
      ls_auart-option = 'EQ'.
      ls_auart-low    = 'ZSTK'.

      APPEND ls_auart TO lr_auart.
      CLEAR  ls_auart.

    WHEN OTHERS.
      REFRESH lr_auart.

  ENDCASE.

  SELECT a~aufnr, a~auart, a~plnbez, a~gamng ,a~meins,
         a~gstrp, a~gltrp, a~plnum, a~plpos, a~vbeln, a~posnr, a~apstat,
        CASE a~auart
          WHEN 'ZSOR' THEN '주문 생산'
          WHEN 'ZFCT' THEN '계획 생산'
          WHEN 'ZSTK' THEN '보충 생산'
          WHEN 'ZCUS' THEN '이월 생산'
          END AS auart_txt,
        CASE a~apstat
          WHEN '1' THEN '승인 하기'
          WHEN '2' THEN '승인 완료'
          ELSE '반려'
          END AS stat_txt,
          CASE a~apstat
          WHEN '1' THEN '@5D@'
          WHEN '2' THEN '@5B@'
          ELSE '@5C@'
          END AS stat_icon,
         b~maktx
    FROM zte2pp0009 AS a
   INNER JOIN zte2mm0001 AS b
      ON a~plnbez = b~matnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_order
   WHERE a~apstat IN @lr_apstat
     AND a~gstrp  IN @lr_gstrp
     AND a~plnbez IN @lr_plnbez
     AND a~auart  IN @lr_auart
   ORDER BY a~gstrp. " 생산 계획일이 빠른 날짜를 상단에 표시

  IF sy-subrc <> 0.
    MESSAGE : s000 WITH '해당 조건의 생산 오더가 존재하지 않습니다.' DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.

FORM create_object .

    EXPORTING

  CREATE OBJECT gcl_grid_order
    EXPORTING
      i_parent = gcl_container_order.

  CREATE OBJECT gcl_container_mrp
    EXPORTING
      container_name = 'GCL_CONTAINER_MRP'.

  CREATE OBJECT gcl_grid_mrp
    EXPORTING
      i_parent = gcl_container_mrp.

  CREATE OBJECT gcl_container_rou
    EXPORTING
      container_name = 'GCL_CONTAINER_ROU'.

  CREATE OBJECT gcl_grid_rou
    EXPORTING
      i_parent = gcl_container_rou.

  CREATE OBJECT gcl_container_doc
    EXPORTING
      container_name = 'GCL_CONTAINER_DOC'.

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container_doc
      rows    = 1
      columns = 2.

  CALL METHOD gcl_splitter->set_column_width
    EXPORTING
      id    = 1      " 왼쪽 칸 번호
      width = 30.    " 퍼센트(%) 단위

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 1
    RECEIVING
      container = gcl_container_lf.

  CREATE OBJECT gcl_document_lf
    EXPORTING
      style            = 'ALV_TO_HTML'
      background_color = 1
      no_margins       = space.

  PERFORM set_document USING gcl_document_lf.

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1
      column    = 2
    RECEIVING
      container = gcl_container_rg.

  CREATE OBJECT gcl_document_rg
    EXPORTING
      style            = 'ALV_TO_HTML'
      background_color = 1
      no_margins       = space.

  PERFORM set_document_2 USING gcl_document_rg.

  gs_variant-report = sy-repid.

  IF gcl_handler IS INITIAL.
    CREATE OBJECT gcl_handler.
  ENDIF.

  SET HANDLER : gcl_handler->handle_hotspot      FOR gcl_grid_order,
                gcl_handler->handle_toolbar      FOR gcl_grid_order,
                gcl_handler->handle_user_command FOR gcl_grid_order,
                gcl_handler->handle_button_click FOR gcl_grid_order.

  CALL METHOD gcl_grid_order->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_order
      it_fieldcatalog = gt_fcat_order.

  CALL METHOD gcl_grid_mrp->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_mrp
      it_fieldcatalog = gt_fcat_mrp.

  CALL METHOD gcl_grid_rou->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant

*&---------------------------------------------------------------------*
*& Form get_mat_req
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_ORDER_MATNR
*&---------------------------------------------------------------------*
FORM get_mat_req  USING pv_matnr TYPE zte2pp0002-matnr
                        pv_gamng TYPE zte2pp0009-gamng
                        pv_gstrp TYPE zte2pp0009-gstrp.

  DATA : lv_vfdat TYPE zte2pp0009-gstrp,
         lv_mhdhb TYPE zte2mm0001-mhdhb.

  SELECT SINGLE mhdhb
    FROM zte2mm0001
   WHERE matnr = @pv_matnr
    INTO @lv_mhdhb.

  lv_vfdat = pv_gstrp + lv_mhdhb + 2.

  REFRESH : gt_mrp, gt_routing.

  SELECT matnr, maktx, menge, meins, unit_meins,
         req_qty_u, req_qty_g, avail_qty_u, avail_qty_g, shortage_u, shortage_g, status, status_txt
    FROM zcds_e2_pp_0003( p_matnr = @pv_matnr,
                          p_gamng = @pv_gamng,
                          p_gstrp = @lv_vfdat )
    INTO CORRESPONDING FIELDS OF TABLE @gt_mrp
    ORDER BY matnr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '생산품의 BOM 이 정의되지 않았습니다' DISPLAY LIKE 'E'..
    EXIT.
  ENDIF.

  SELECT vornr, ktext, ltxa1,
       " CDS의 FLTP 필드를 여기서 바로 DEC로 변환
      CAST( req_time AS DEC( 13, 2 ) ) AS req_time,
       vgw02, vge02, bmsch, meins
  FROM zcds_e2_pp_0004( p_matnr = @pv_matnr, p_gamng = @pv_gamng )
  INTO CORRESPONDING FIELDS OF TABLE @gt_routing
  ORDER BY vornr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '생산품의 공정이 정의되지 않았습니다' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

ENDFORM.

*& Form set_document
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GCL_DOCUMENT_DOC
*&---------------------------------------------------------------------*
FORM set_document  USING    pcl_document TYPE REF TO cl_dd_document.

  DATA : lcl_table   TYPE REF TO cl_dd_table_element,
         lcl_col1    TYPE REF TO cl_dd_area,
         lcl_col2    TYPE REF TO cl_dd_area,
         lcl_col3    TYPE REF TO cl_dd_area,
         lcl_col4    TYPE REF TO cl_dd_area,

         lv_text     TYPE sdydo_text_element,

         lv_wait_cnt TYPE i,
         lv_appr_cnt TYPE i,
         lv_rejt_cnt TYPE i.

  CLEAR : lv_wait_cnt, lv_appr_cnt, lv_rejt_cnt, lv_text.

  LOOP AT gt_order INTO gs_order.

    CASE gs_order-apstat.
      WHEN '1'.
        lv_wait_cnt += 1. " 대기
      WHEN '2'.
        lv_appr_cnt += 1. " 승인
      WHEN '3'.
        lv_rejt_cnt += 1. " 반려
    ENDCASE.

  ENDLOOP.

  lv_text = '생산오더 승인 상태'.
  CALL METHOD pcl_document->add_text
    EXPORTING
      text         = lv_text
      sap_fontsize = cl_dd_document=>large
      sap_emphasis = cl_dd_area=>strong
      sap_color    = cl_dd_area=>list_background_inv.

  CALL METHOD pcl_document->new_line.
  CALL METHOD pcl_document->new_line.

  CALL METHOD pcl_document->add_table
    EXPORTING
      no_of_columns               = 4
      cell_background_transparent = 'X'
      with_heading                = space
      border                      = '0'
    IMPORTING
      table                       = lcl_table.

  CALL METHOD lcl_table->add_column
    EXPORTING
      width  = '20'
    IMPORTING
      column = lcl_col1.

  CALL METHOD lcl_table->add_column
    EXPORTING
      width  = '70'
    IMPORTING
      column = lcl_col2.

  CALL METHOD lcl_table->add_column
    EXPORTING
      width  = '10'
    IMPORTING
      column = lcl_col3.

  CALL METHOD lcl_table->add_column
    EXPORTING
      width  = '20'
    IMPORTING
      column = lcl_col4.

  CALL METHOD lcl_table->new_row.

  CALL METHOD lcl_col1->add_icon
    EXPORTING
      sap_icon = 'ICON_LED_YELLOW'.

  lv_text = '대기:  '.
  CALL METHOD lcl_col2->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.


  lv_text = lv_wait_cnt.
  CALL METHOD lcl_col3->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.

  lv_text = '건'.
  CALL METHOD lcl_col4->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.

  CALL METHOD lcl_table->new_row.

  CALL METHOD lcl_col1->add_icon
    EXPORTING
      sap_icon = 'ICON_LED_GREEN'.

  lv_text = '승인: ' .
  CALL METHOD lcl_col2->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.


  lv_text = lv_appr_cnt.
  CALL METHOD lcl_col3->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.

  lv_text = '건'.
  CALL METHOD lcl_col4->add_text
    EXPORTING
      text         = lv_text
      sap_emphasis = cl_dd_area=>strong.

  CALL METHOD lcl_table->new_row.

  CALL METHOD lcl_col1->add_icon
    EXPORTING
      sap_icon = 'ICON_LED_RED'.

*& Form create_production_order
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> PS_ORDER
*&      --> GT_MRP
*&---------------------------------------------------------------------*
FORM create_production_order  USING  ps_order LIKE gs_order
                                     pt_mrp   LIKE gt_mrp.


  DATA: lt_stock      TYPE TABLE OF zte2mm0004, " 재고 테이블
        ls_stock      LIKE LINE OF lt_stock,
        lt_stock_save LIKE TABLE OF ls_stock,   " 재고테이블 데이터 저장용
        lv_needed_qty TYPE zte2pp0010-bdmng,
        lv_alloc_qty  TYPE zte2pp0010-bdmng,
        ls_item       TYPE zte2pp0010,        " 생산 아이템 테이블
        lt_item       LIKE TABLE OF ls_item,
        ls_reser      TYPE zte2mm0017,        " 예약 테이블
        lt_reser      LIKE TABLE OF ls_reser,
        lv_rsnum      TYPE zte2mm0017-rsnum,  " 예약 번호
        lv_tabix      TYPE sy-tabix,
        lv_posnr      TYPE zte2pp0010-posnr, " 생산 오더 항목번호
        ls_mov_h      TYPE zte2mm0011,
        ls_mov_i      TYPE zte2mm0012,
        lt_mov_i      LIKE TABLE OF ls_mov_i,
        ls_timestamp  TYPE zse2timestamp,
        lv_commit,
        lv_err,
        BEGIN OF ls_recp,
          plnnr TYPE zte2pp0009-plnnr,
          stlnr TYPE zte2pp0009-stlnr,
        END OF ls_recp,
        lv_mhdhb TYPE zte2mm0001-mhdhb,
        lv_vfdat TYPE zte2pp0009-gstrp.

  CLEAR : lv_posnr, lv_tabix, ls_item, ls_stock, lv_needed_qty, lv_alloc_qty.
  REFRESH : lt_item, lt_stock_save, lt_stock, lt_reser.

* BOM, 라우팅 연결
  SELECT SINGLE plnnr, stlnr
    FROM zte2pp0008
   WHERE matnr = @ps_order-plnbez
    INTO CORRESPONDING FIELDS OF @ls_recp.

*       생산 승인 상태 변경
  ps_order-apstat = '2'.

* 자재 예약 번호 채번
  PERFORM get_rs_num USING lv_rsnum.


* 자재이동문서 헤더, 아이템
  ls_mov_h = VALUE #(
    mjahr = sy-datum(4)
    aufnr = ps_order-aufnr
    bldat = sy-datum
    budat = sy-datum
    bukrs = '1000'
  ).

* 투입 자재 유통기한
  SELECT SINGLE mhdhb
    FROM zte2mm0001
   WHERE matnr = @ps_order-plnbez
    INTO @lv_mhdhb.

  " 생산품의 보존기한 + 납기시간 + 고객사 최소 보장기간
  lv_vfdat = ps_order-gstrp + lv_mhdhb + 2.

  " 생산시 들어가는 자재 아이템에 저장
  " 생산에 필요한 자재별로 루프
  LOOP AT pt_mrp INTO gs_mrp.
*    lv_needed_qty = gs_mrp-req_qty_g. " 이번 생산에 필요 수량
    lv_needed_qty = gs_mrp-req_qty_u. " 이번 생산에 필요 수량 --> 이제 단위 변경

    " 해당 자재의 가용재고 유통기한 순으로 가져오기 (같은배치는 같은 유통기한)
    SELECT matnr charg  lgort  werks  mtart
           labst insme speme retme vfdat
           meins dmbtr waers
           ernam erdat erzet aenam aedat aezet
      INTO CORRESPONDING FIELDS OF TABLE lt_stock
      FROM zte2mm0004
     WHERE matnr = gs_mrp-matnr
       AND labst > 0 " 재고가 존재할때만
       AND vfdat >= lv_vfdat " 생산예정일 이후의 유통기한이 남아있는재고만
     ORDER BY vfdat ASCENDING.

    LOOP AT lt_stock INTO ls_stock.

      lv_tabix = sy-tabix.

      IF lv_needed_qty <= 0. " 다 채웠으면 탈출
        EXIT.
      ENDIF.

      " 특정 배치에서 가져올 수량 결정
      IF ls_stock-labst >= lv_needed_qty. " 배치의 재고가 충분하면 ?
        lv_alloc_qty = lv_needed_qty. " 전부 가져감
      ELSE.
        lv_alloc_qty = ls_stock-labst. " 있는거 전부
      ENDIF.

**********************************************************************
*                     데이터 가공, 테이블 추가용 itab
**********************************************************************
*     1. 재고 차감 기록
*      MOVE-CORRESPONDING ls_stock TO ls_timestamp.
*
*      CALL FUNCTION 'ZFE2US0002'
*        EXPORTING
*          is_timestamp = ls_timestamp
*        IMPORTING
*          es_timestamp = ls_timestamp.
*
*      MOVE-CORRESPONDING ls_timestamp TO ls_stock.
*
*      ls_stock-labst = ls_stock-labst - lv_alloc_qty.
*
*      APPEND  ls_stock TO lt_stock_save.
*      CLEAR : ls_timestamp.

      CALL FUNCTION 'ZFE2US0002'
        EXPORTING
          is_timestamp = ls_timestamp
        IMPORTING
          es_timestamp = ls_timestamp.

      lv_posnr += 10.
      CLEAR ls_item.

*     1. 자재이동문서 아이템

*      ls_mov_i-mjahr = sy-datum(4).
      ls_mov_i-bwart = '301'. " 플랜트 간 이전
      ls_mov_i-charg = ls_stock-charg. " 배치
      ls_mov_i-matnr = ls_stock-matnr. " 자재
      ls_mov_i-menge = lv_alloc_qty. " 몇개
      ls_mov_i-meins = ls_stock-meins. "단위
      ls_mov_i-werks = ls_stock-werks. "플랜트
      ls_mov_i-umwrk = '1000'. " 이동후 플랜트
      ls_mov_i-lgort = ls_stock-lgort. "저장위치
      ls_mov_i-waers = 'KRW'.
      ls_mov_i-umlgo = 'PP01'. " 이동후 lgort
      ls_mov_i-posnr = lv_posnr.
      ls_mov_i-btext = '생산투입 전 자재이동'.

      APPEND ls_mov_i TO lt_mov_i.
      CLEAR  ls_mov_i.

*     2. 생산오더 아이템 데이터 구성
      ls_item-aufnr = ps_order-aufnr. " 생산오더번호
      ls_item-posnr = lv_posnr.       " 항목번호
      ls_item-matnr = gs_mrp-matnr.   " 자재번호
      ls_item-charg = ls_stock-charg. " 배치
      ls_item-bdmng = lv_alloc_qty.   " 수량

FORM f4_pernr .

          pernr TYPE zte2hr0002-pernr,
          sname TYPE zte2hr0002-sname, " 사원 성명
          orgtx TYPE zte2hr0001-orgtx, " 부서명
        END OF ls_pernr,

        lt_return TYPE TABLE OF ddshretval,
        ls_return TYPE ddshretval,
        lt_dynp   TYPE TABLE OF dynpread,
        ls_dynp   TYPE dynpread.

    FROM zte2hr0002 AS a
   INNER JOIN zte2hr0001 AS b
      ON a~orgeh = b~orgeh
     AND a~orgeh = '30000000'
     INTO CORRESPONDING FIELDS OF TABLE @lt_pernr.


  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING

      dynpprog     = sy-repid

      value_org    = 'S'
    TABLES
      value_tab    = lt_pernr
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INTO ls_return INDEX 1.

    IF sy-subrc = 0.

      READ TABLE lt_pernr INTO ls_pernr WITH KEY pernr = ls_return-fieldval.

      IF sy-subrc = 0.
