MODULE get_pressed INPUT.

  CASE gv_okcode.
    WHEN 'MRP'.
      gv_pressedtab = gv_okcode.

      CLEAR gv_okcode.

    WHEN 'ROUTING'.
      gv_pressedtab = gv_okcode.

      CLEAR gv_okcode.

  ENDCASE.

ENDMODULE.

*&      Module  USER_COMMAND_0100  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE user_command_0100 INPUT.

  CASE gv_okcode.

    WHEN 'REFRESH'.
      CLEAR gv_okcode.
      CLEAR gs_search.

      PERFORM init_param.
      PERFORM get_order.
      PERFORM set_style.
      PERFORM refresh_grid_order.
      PERFORM refresh_doc_l.

      REFRESH : gt_mrp, gt_routing.
      PERFORM refresh_grid_mrp.
      PERFORM refresh_grid_rou.

    WHEN 'SEARCH'.
      DATA    lv_err.
      CLEAR   gv_okcode.
      PERFORM check_valid_date CHANGING lv_err.
      CHECK   lv_err IS INITIAL.

      PERFORM get_order.
      PERFORM set_style.
      PERFORM refresh_grid_order.
      PERFORM refresh_doc_l.

      REFRESH : gt_mrp, gt_routing.
      PERFORM refresh_grid_mrp.
      PERFORM refresh_grid_rou.

    WHEN 'RADIO_BTN'.
      CLEAR gv_okcode.


  ENDCASE.

ENDMODULE.

MODULE user_command_0103 INPUT.

  CASE gv_okcode.
    WHEN 'OKAY'. " 담당자 팝업
      PERFORM check_valid_pernr.
      IF gv_pernr_invalid = 'X'..
        EXIT.
      ENDIF.

      CASE gv_ord_type.
        WHEN 'R'.
          " PR 생성
          PERFORM create_purchase_request USING gs_order.

          " 메인화면 데이터, 그리드 갱신
          PERFORM get_order.
          PERFORM set_style.
          PERFORM refresh_grid_order.
          PERFORM refresh_doc_l.

          REFRESH : gt_mrp, gt_routing.
          PERFORM refresh_grid_mrp.
          PERFORM refresh_grid_rou.

          CLEAR gs_pernr.
          LEAVE TO SCREEN 0. " 팝업 닫기

        WHEN 'S'.
          PERFORM create_production_order USING  gs_order
                                                 gt_mrp.

          PERFORM get_order.
          PERFORM set_style.
          PERFORM refresh_grid_order.
          PERFORM refresh_doc_l.

          REFRESH : gt_mrp, gt_routing.
          PERFORM refresh_grid_mrp.
          PERFORM refresh_grid_rou.

          CLEAR gs_pernr.
          LEAVE TO SCREEN 0.

      ENDCASE.

  ENDCASE.

ENDMODULE.
