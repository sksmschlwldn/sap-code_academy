*&---------------------------------------------------------------------*
*& Form set_popor
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_month .

  REFRESH gt_poper_list.

  " 로컬 날짜(sy-datlo)에서 현재 월만 바로 추출
  DATA(lv_current_month) = sy-datlo+4(2).

  " 이번 달까지만 리스트 생성
  DO lv_current_month TIMES.
    CLEAR gs_poper_list.

    " Key: 001, 002... (DB 저장용)
    gs_poper_list-key  = |{ sy-index WIDTH = 3 ALIGN = RIGHT PAD = '0' }|.

    " Text: 1월, 2월... (화면 표시용)
    gs_poper_list-text = |{ sy-index }월|.

    APPEND gs_poper_list TO gt_poper_list.
  ENDDO.

  " S_MON 변수값이 비어있다면 초기값 설정
  IF gs_param-s_mon IS INITIAL.
    gs_param-s_mon = |{ lv_current_month WIDTH = 3 ALIGN = RIGHT PAD = '0' }|.
  ENDIF.

  " 기존 E_MON 로직은 그대로 유지
  DATA(lv_now2) = sy-datlo+4(2).
  gs_param-e_mon = |0{ lv_now2 }|.

  " 리스트박스 세팅 (S_MON만 수행)
  CALL FUNCTION 'VRM_SET_VALUES'
    EXPORTING
      id     = 'GS_PARAM-S_MON'
      values = gt_poper_list.

ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_year .

* DATA: lt_year_list TYPE vrm_values,
* ls_year_list TYPE vrm_value.
* DATA: lv_end_year   TYPE i VALUE 1996,
* lv_start_year TYPE i.
*
* lv_start_year = sy-datlo(4).
*
* REFRESH lt_year_list.
*
* " 1996년부터 현재 연도까지 루프
* WHILE lv_end_year <= lv_start_year.
* CLEAR ls_year_list.
*
* ls_year_list-key  = |{ lv_start_year }|.
*
* ls_year_list-text = |{ lv_start_year }년|.
*
* APPEND ls_year_list TO lt_year_list.
* lv_start_year = lv_start_year - 1.
* ENDWHILE.
*
* " 리스트박스 세팅
* CALL FUNCTION 'VRM_SET_VALUES'
* EXPORTING
* id     = 'GS_PARAM-GJAHR'  " 화면에 선언된 변수명
* values = lt_year_list.

  gs_param-gjahr = sy-datlo(4).
  gv_e_mon       = |{ sy-datlo+4(2) }월|.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form get_segment_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_segment_data .

  REFRESH gt_segment.

  SELECT c13~segm_no c13~segm_name c13~s_kostl c13~s_hkont_low c13~s_hkont_high
         c13~r_kostl c13~r_prctr   c13~kstar   c13~al_skf_id

         c2~ktext AS r_prctr_txt

         c3_1~ktext AS s_kostl_txt
         c3_2~ktext AS r_kostl_txt

         c10~ktext AS skf_txt

         f1~txt20 AS kstar_txt

    INTO CORRESPONDING FIELDS OF TABLE gt_segment
    FROM zte4co0013 AS c13

    LEFT OUTER JOIN zte4co0002 AS c2
      ON c13~r_prctr = c2~prctr

    LEFT OUTER JOIN zte4co0003 AS c3_1
      ON c13~s_kostl = c3_1~kostl

    LEFT OUTER JOIN zte4co0003 AS c3_2
      ON c13~r_kostl = c3_2~kostl

    LEFT OUTER JOIN zte4co0010 AS c10
      ON c13~al_skf_id = c10~stagr

    LEFT OUTER JOIN zte4fi0001 AS f1
      ON c13~kstar = f1~saknr

   WHERE cycle =  gs_cycle-cycle
     AND sdate =  gs_cycle-sdate
     AND lvorm <> 'X'

ORDER BY segm_no.

  IF sy-subrc NE 0.
    MESSAGE s033 WITH '해당 사이클의 세그먼트' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form calculate_allocation_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM calculate_allocation_data .

  DATA: lt_segments      LIKE TABLE OF gs_segment
                         WITH EMPTY KEY,
        lv_total_skf     TYPE zte4co0011-stsvl,    " 전체 SKF 합계
        lv_recv_skf      TYPE zte4co0011-stsvl,    " 개별 리시버 SKF
        lv_meins         TYPE zte4co0010-msehi,    " SFK 단위
        lv_prev_alloc    TYPE zte4fi0003-wsl,      " 기배부액
        lv_target_amt    TYPE zte4fi0003-wsl,      " 구간 전체 목표액
        lv_current_alloc TYPE zte4fi0003-wsl,      " 당월 최종 배부액
        lv_perc          TYPE p DECIMALS 2,        " 배부 비율 (%)
        lv_prev_mon      TYPE monat.               " 기배부액 조회용

  " 세그먼트 정보 일괄 조회
  IF gt_selected_cycle IS INITIAL.
    MESSAGE s036 WITH '배부 사이클' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SELECT c13~cycle, c13~sdate, c13~segm_no, c13~segm_name,
         c13~s_kostl, c13~s_hkont_low, c13~s_hkont_high,
         c13~r_kostl, c13~r_prctr, c13~kstar, c13~al_skf_id,

         c2~ktext AS r_prctr_txt,

         c3_1~ktext AS s_kostl_txt,
         c3_2~ktext AS r_kostl_txt,

         c10~ktext AS skf_txt,

         f1_1~txt20 AS s_hkont_low_txt,
         f1_2~txt20 AS s_hkont_high_txt,
         f1_3~txt20 AS kstar_txt

    FROM zte4co0013 AS c13

   INNER JOIN zte4co0002 AS c2
      ON c13~r_prctr = c2~prctr

   INNER JOIN zte4co0003 AS c3_1
      ON c13~s_kostl = c3_1~kostl

   INNER JOIN zte4co0003 AS c3_2
      ON c13~r_kostl = c3_2~kostl

   INNER JOIN zte4co0010 AS c10
      ON c13~al_skf_id = c10~stagr

   INNER JOIN zte4fi0001 AS f1_1
      ON c13~s_hkont_low = f1_1~saknr

   INNER JOIN zte4fi0001 AS f1_2
      ON c13~s_hkont_high = f1_2~saknr

   INNER JOIN zte4fi0001 AS f1_3
      ON c13~kstar = f1_3~saknr

   INNER JOIN @gt_selected_cycle AS sel
      ON c13~cycle = sel~cycle
     AND c13~sdate = sel~sdate

    INTO CORRESPONDING FIELDS OF TABLE @lt_segments.

  IF lt_segments IS INITIAL.
    MESSAGE s033 WITH '등록된 배부 세그먼트' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  " 배부 대상 코스트센터별 순수 비용 잔액 조회
  SELECT CASE
           WHEN f3~drcrk = 'S' AND f3~r_kostl NE @space THEN f3~r_kostl
           ELSE f3~s_kostl
         END AS s_kostl,
         f3~racct,
         f1~txt20 AS racct_txt,
         SUM( CAST( CASE f3~drcrk
                      WHEN 'S' THEN f3~wsl
                      WHEN 'H' THEN ( f3~wsl * -1 )
                      ELSE 0
                    END AS DEC( 31, 2 ) ) ) AS total_wsl
     INTO CORRESPONDING FIELDS OF TABLE @gt_fi_base_costs
     FROM zte4fi0003 AS f3
    INNER JOIN zte4fi0002 AS f2 ON f3~rbukrs = f2~bukrs
                               AND f3~gjahr  = f2~gjahr
                               AND f3~belnr  = f2~belnr
    INNER JOIN zte4fi0001 AS f1 ON f3~racct  = f1~saknr
    WHERE f2~bukrs   = '0001'
      AND f2~gjahr   = @gs_param-gjahr
      AND f2~monat   = @gs_param-e_mon
      AND f2~blart  <> 'CV'
      AND ( f3~s_kostl NE '' OR f3~r_kostl NE '')
      " 제조비/판관비
      AND f3~racct   BETWEEN '700000' AND '899999'
    GROUP BY CASE
               WHEN f3~drcrk = 'S' AND f3~r_kostl NE @space THEN f3~r_kostl
               ELSE f3~s_kostl
             END,
             f3~racct, f1~txt20.

  DELETE gt_fi_base_costs WHERE total_wsl = 0.

  REFRESH gt_result.

  gt_result = VALUE #(
      FOR ls_seg IN lt_segments
      FOR ls_costs IN gt_fi_base_costs WHERE ( s_kostl = ls_seg-s_kostl
                                           AND racct  >= ls_seg-s_hkont_low
                                           AND racct  <= ls_seg-s_hkont_high
                                           AND total_wsl <> 0 ) (
        cycle            = ls_seg-cycle
        segm_no          = ls_seg-segm_no
        segm_name        = ls_seg-segm_name
        s_kostl          = ls_seg-s_kostl
        s_kostl_txt      = ls_seg-s_kostl_txt
        s_hkont_low      = ls_costs-racct
        s_hkont_low_txt  = ls_costs-racct_txt
        s_hkont_high     = ls_costs-racct
        s_hkont_high_txt = ls_costs-racct_txt
        send_amt         = ls_costs-total_wsl
        r_kostl          = ls_seg-r_kostl
        r_kostl_txt      = ls_seg-r_kostl_txt
        r_prctr          = ls_seg-r_prctr
        r_prctr_txt      = ls_seg-r_prctr_txt
        kstar            = ls_seg-kstar
        kstar_txt        = ls_seg-kstar_txt
        al_skf_id        = ls_seg-al_skf_id
        skf_txt          = ls_seg-skf_txt
        waers            = 'KRW'
        alloc_type       = COND #( WHEN ls_seg-r_kostl IS INITIAL THEN 'PC' ELSE 'CC' )
      )
    ).

  DATA: BEGIN OF ls_seg_total,
          cycle        TYPE zte4co0013-cycle,
          s_kostl      TYPE zte4co0013-s_kostl,
          s_hkont_low  TYPE zte4co0013-s_hkont_low,
          s_hkont_high TYPE zte4co0013-s_hkont_high,
          total_wsl    TYPE zte4fi0003-wsl,
        END OF ls_seg_total,
        lt_seg_total LIKE TABLE OF ls_seg_total.

  lt_seg_total = VALUE #(
    FOR GROUPS <grp1> OF <ls_row> IN gt_result
    GROUP BY ( cycle        = <ls_row>-cycle
               s_kostl      = <ls_row>-s_kostl
               s_hkont_low  = <ls_row>-s_hkont_low
               s_hkont_high = <ls_row>-s_hkont_high
               send_amt     = <ls_row>-send_amt )
    ( cycle        = <grp1>-cycle
      s_kostl      = <grp1>-s_kostl
      s_hkont_low  = <grp1>-s_hkont_low
      s_hkont_high = <grp1>-s_hkont_high
      total_wsl    = REDUCE zte4fi0003-wsl(
                       INIT lv_sum = VALUE zte4fi0003-wsl( )
                       FOR ls_s IN lt_segments WHERE ( cycle = <grp1>-cycle AND s_kostl = <grp1>-s_kostl
                           AND s_hkont_low <= <grp1>-s_hkont_low AND s_hkont_high >= <grp1>-s_hkont_high )
                       FOR ls_c IN gt_fi_base_costs WHERE ( s_kostl = ls_s-s_kostl
                                                       AND racct  >= ls_s-s_hkont_low
                                                       AND racct  <= ls_s-s_hkont_high )
                       NEXT lv_sum = lv_sum + ls_c-total_wsl ) ) ).

  " 동일 기간에 이미 실행된 같은 배부사이클 금액까지 기배부액으로 본다.
  " 그래야 같은 사이클을 재실행했을 때 recv_amt = 0 이 되어 중복 전표가 생성되지 않는다.
  SELECT seg~cycle,
         seg~segm_no,
         f3~s_kostl,
         f3~r_kostl,
         f3~racct,
         SUM( f3~wsl ) AS prev_wsl

    FROM zte4fi0003 AS f3

   INNER JOIN zte4fi0002 AS f2
      ON f3~rbukrs = f2~bukrs
     AND f3~gjahr  = f2~gjahr
     AND f3~belnr  = f2~belnr

   INNER JOIN @lt_segments AS seg
      ON f3~s_kostl = seg~s_kostl
     AND f3~r_kostl = seg~r_kostl
     AND f3~racct   = seg~kstar

   WHERE f2~bukrs = '0001'
     AND f2~gjahr = @gs_param-gjahr
     AND f2~monat BETWEEN @gs_param-s_mon AND @gs_param-e_mon
     AND f2~blart = 'CA'
     AND f3~drcrk = 'S'

   GROUP BY seg~cycle,
            seg~segm_no,
            f3~s_kostl,
            f3~r_kostl,
            f3~racct

    INTO TABLE @DATA(lt_prev_alloc).

  DATA: BEGIN OF ls_skf_driver,
          al_skf_id TYPE zee4_co_stagr,
          r_kostl   TYPE zee4_co_kostl,
        END OF ls_skf_driver,
        lt_skf_driver LIKE TABLE OF ls_skf_driver WITH EMPTY KEY.

  lt_skf_driver = VALUE #( FOR <ls_res> IN gt_result
                             ( al_skf_id = <ls_res>-al_skf_id
                               r_kostl   = <ls_res>-r_kostl ) ).

  SORT lt_skf_driver BY al_skf_id r_kostl.
  DELETE ADJACENT DUPLICATES FROM lt_skf_driver COMPARING al_skf_id r_kostl.

  SELECT f11~stagr, f11~kostl, SUM( f11~stsvl ) AS stsvl, c10~msehi AS meins

    FROM zte4co0011 AS f11

   INNER JOIN @lt_skf_driver AS drv
      ON f11~stagr = drv~al_skf_id
     AND f11~kostl = drv~r_kostl

   INNER JOIN zte4co0010 AS c10
      ON f11~stagr = c10~stagr

   WHERE f11~gjahr = @gs_param-gjahr
     AND f11~poper BETWEEN @gs_param-s_mon AND @gs_param-e_mon

GROUP BY f11~stagr, f11~kostl, c10~msehi
ORDER BY f11~stagr, f11~kostl, c10~msehi

     INTO TABLE @DATA(lt_skf_data).

  DATA : lt_receiver_skf_sum LIKE TABLE OF gs_result.

  lt_receiver_skf_sum = VALUE #(
    FOR GROUPS <grp2> OF <ls_res> IN gt_result
    GROUP BY ( cycle        = <ls_res>-cycle
               s_kostl      = <ls_res>-s_kostl
               s_hkont_low  = <ls_res>-s_hkont_low
               s_hkont_high = <ls_res>-s_hkont_high
               al_skf_id    = <ls_res>-al_skf_id )

    ( cycle        = <grp2>-cycle
      s_kostl      = <grp2>-s_kostl
      s_hkont_low  = <grp2>-s_hkont_low
      s_hkont_high = <grp2>-s_hkont_high
      al_skf_id    = <grp2>-al_skf_id

      total_skf = REDUCE #( INIT val2 = VALUE zee4_co_stsvl( )
                            FOR m2 IN GROUP <grp2>
                            NEXT val2 = val2 + VALUE #( lt_skf_data[ stagr = m2-al_skf_id
                                                                     kostl = m2-r_kostl ]-stsvl DEFAULT 0 ) ) ) ) .

  SORT lt_skf_data BY stagr kostl.

  LOOP AT gt_result ASSIGNING FIELD-SYMBOL(<fs_res>).
    CLEAR: lv_total_skf, lv_recv_skf, lv_prev_alloc.

    READ TABLE lt_receiver_skf_sum INTO DATA(ls_tot) WITH KEY cycle        = <fs_res>-cycle
                                                              s_kostl      = <fs_res>-s_kostl
                                                              s_hkont_low  = <fs_res>-s_hkont_low
                                                              s_hkont_high = <fs_res>-s_hkont_high.

    IF sy-subrc = 0.
      lv_total_skf = ls_tot-total_skf.
    ENDIF.

    READ TABLE lt_skf_data INTO DATA(ls_ind) WITH KEY stagr = <fs_res>-al_skf_id
                                                      kostl = <fs_res>-r_kostl
                                                      BINARY SEARCH.

    IF sy-subrc = 0.
      lv_recv_skf = ls_ind-stsvl.
      lv_meins    = ls_ind-meins.
    ENDIF.

    IF ( lv_total_skf <= 0 OR lv_recv_skf <= 0 )
   AND gv_exit_program = abap_true.

      gv_error = 'X'.
      MESSAGE i001 WITH 'SKF 기준값 또는 수신처 실적이 누락되어 계산을 중단합니다.'.
      RETURN.

    ENDIF.

    READ TABLE lt_prev_alloc INTO DATA(ls_prev) WITH KEY cycle   = <fs_res>-cycle
                                                         segm_no = <fs_res>-segm_no
                                                         s_kostl = <fs_res>-s_kostl
                                                         r_kostl = <fs_res>-r_kostl
                                                         racct   = <fs_res>-kstar.

    IF sy-subrc = 0.
      READ TABLE lt_seg_total INTO DATA(ls_total) WITH KEY cycle        = <fs_res>-cycle
                                                           s_kostl      = <fs_res>-s_kostl
                                                           s_hkont_low  = <fs_res>-s_hkont_low
                                                           s_hkont_high = <fs_res>-s_hkont_high.

      IF ls_total-total_wsl <> 0.
        lv_prev_alloc = ls_prev-prev_wsl * ( <fs_res>-send_amt / ls_total-total_wsl ).
      ENDIF.

    ENDIF.

    <fs_res>-target_amt = ( <fs_res>-send_amt * lv_recv_skf ) / lv_total_skf.
    <fs_res>-prev_amt   = lv_prev_alloc.
    <fs_res>-recv_amt   = <fs_res>-target_amt - lv_prev_alloc.
    <fs_res>-skf_perc   = ( lv_recv_skf / lv_total_skf ) * 100.
    <fs_res>-skf_val    = lv_recv_skf.
    <fs_res>-total_skf  = lv_total_skf.
    <fs_res>-meins      = lv_meins.

  ENDLOOP.

  DELETE gt_result WHERE recv_amt <= 0.

  DATA: lv_diff TYPE zte4fi0003-wsl.

  SELECT r~s_kostl, r~s_hkont_low, r~s_hkont_high, r~send_amt,
         MAX( r~r_kostl_txt ) AS last_r_txt,
         SUM( r~recv_amt )    AS sum_recv
    FROM @gt_result AS r
   GROUP BY r~s_kostl, r~s_hkont_low, r~s_hkont_high, r~send_amt
    INTO TABLE @DATA(lt_group_diffs).

  LOOP AT lt_group_diffs INTO DATA(ls_diff).

    lv_diff = ls_diff-send_amt - ls_diff-sum_recv.

    IF lv_diff = 0.
      CONTINUE.
    ENDIF.

    READ TABLE gt_result ASSIGNING FIELD-SYMBOL(<fs_last>)
      WITH KEY s_kostl      = ls_diff-s_kostl
               s_hkont_low  = ls_diff-s_hkont_low
               s_hkont_high = ls_diff-s_hkont_high
               r_kostl_txt  = ls_diff-last_r_txt.

    IF sy-subrc = 0.
      <fs_last>-recv_amt   = <fs_last>-recv_amt + lv_diff.
      <fs_last>-target_amt = <fs_last>-target_amt + lv_diff.
    ENDIF.

  ENDLOOP.

  PERFORM check_allocation_error CHANGING gt_result gt_result_error gv_error.

ENDFORM.

*& Form handle_execution_mode
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM handle_execution_mode .

  DATA: lv_allow_extract TYPE abap_bool VALUE abap_true.

  IF gt_result IS INITIAL.
    MESSAGE s052 WITH '배부 처리 대상' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  IF gv_error IS NOT INITIAL.

    CASE 'X'.
      WHEN gv_simulation_mode.
        PERFORM display_error_popup.
        lv_allow_extract = abap_true.
        CLEAR gv_error.
      WHEN OTHERS.
        CLEAR gv_error.
        EXIT.
    ENDCASE.
  ELSE.

    IF gv_test = 'X'.
      MESSAGE s001 WITH '테스트 실행입니다. 전표가 생성되지 않습니다.' DISPLAY LIKE 'S'.
    ELSE.
      PERFORM post_allocation_results.
    ENDIF.

  ENDIF.

  IF gv_save_extract = abap_true AND gv_extract_id IS NOT INITIAL.
    PERFORM save_allocation_extract.
  ENDIF.

  PERFORM display_alv_result.

ENDFORM.

        ON f03~s_kostl = c3_1~kostl

     LEFT OUTER JOIN zte4co0003 AS c3_2

        ON f03~racct = f1_1~saknr

     LEFT OUTER JOIN zte4fi0001 AS f1_3
        ON f03~racct = f1_3~saknr

     INTO CORRESPONDING FIELDS OF TABLE @gt_journal.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_hierarchy
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      <-- GT_TREE_NODE
*&---------------------------------------------------------------------*
FORM make_hierarchy.

  DATA: lv_root_key  TYPE lvc_nkey,
        lv_start_key TYPE lvc_nkey.

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE gt_hier
    FROM zte4co0007
   WHERE setnam = 'CCSH1000'.

  LOOP AT gt_hier ASSIGNING FIELD-SYMBOL(<fs_h>) WHERE node_type = 'L'.
    READ TABLE gt_result TRANSPORTING NO FIELDS WITH KEY s_kostl = <fs_h>-child_id.
    IF sy-subrc <> 0.
      <fs_h>-parent_id = 'DELETE'.
    ENDIF.
  ENDLOOP.

  DELETE gt_hier WHERE parent_id = 'DELETE'.
  SORT gt_hier BY hlevel DESCENDING.

  LOOP AT gt_hier ASSIGNING FIELD-SYMBOL(<fs_g>) WHERE node_type = 'G'.
    READ TABLE gt_hier TRANSPORTING NO FIELDS WITH KEY parent_id = <fs_g>-child_id.
    IF sy-subrc <> 0.
      <fs_g>-parent_id = 'DELETE'.
    ENDIF.
  ENDLOOP.

  DELETE gt_hier WHERE parent_id = 'DELETE'.
  SORT gt_hier BY hlevel ASCENDING.

  CLEAR lv_root_key.

  LOOP AT gt_hier INTO DATA(ls_root) WHERE parent_id IS INITIAL OR hlevel = 1.
    PERFORM add_recursive_node USING ls_root
                                     lv_root_key
                                     gt_hier
                            CHANGING lv_start_key.

    IF lv_start_key IS NOT INITIAL.
      CALL METHOD go_alv_tree->expand_node
        EXPORTING
          i_node_key    = lv_start_key
          i_level_count = 99
        EXCEPTIONS
          OTHERS        = 0.
    ENDIF.
  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form add_recursive_node
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
FORM add_recursive_node USING    ps_hier    LIKE gs_hier
                                 pv_p_key   TYPE lvc_nkey
                                 pt_hier    LIKE gt_hier
                        CHANGING cv_new_key TYPE lvc_nkey.

  DATA: ls_display LIKE gs_tree_node,
        lv_sub_key TYPE lvc_nkey.

  CLEAR ls_display.
  ls_display-text     = ps_hier-node_text.
  ls_display-node_key = ps_hier-child_id.

  PERFORM add_node USING pv_p_key
                         ps_hier-node_type
                CHANGING ls_display.

  cv_new_key = ls_display-nkey.

  READ TABLE gt_hier ASSIGNING FIELD-SYMBOL(<fs_h>)
       WITH KEY child_id = ps_hier-child_id.
  IF sy-subrc = 0.
    <fs_h>-nkey = ls_display-nkey.
  ENDIF.

  IF ps_hier-node_type = 'G'.
    LOOP AT pt_hier INTO DATA(ls_child) WHERE parent_id = ps_hier-child_id.
      PERFORM add_recursive_node USING ls_child
                                       ls_display-nkey
                                       pt_hier
                              CHANGING lv_sub_key.
    ENDLOOP.
  ENDIF.

ENDFORM.

FORM handle_item_double_click  USING pv_node_key  TYPE lvc_nkey
                                     pv_fieldname TYPE lvc_fname.

  IF pv_fieldname NE '&Hierarchy'.
    EXIT.
  ENDIF.

  READ TABLE gt_hier INTO DATA(ls_clicked) WITH KEY nkey = pv_node_key.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CLEAR gt_kostl_filter.
  gs_kostl_filter = VALUE #( fieldname = 'S_KOSTL' sign = 'I' option = 'EQ' ).

  IF ls_clicked-node_type = 'L'.
    gs_kostl_filter-low = ls_clicked-child_id.
    APPEND gs_kostl_filter TO gt_kostl_filter.
  ELSEIF ls_clicked-node_type = 'G'.
    PERFORM collect_childs USING    ls_clicked-child_id
                           CHANGING gt_kostl_filter.
  ENDIF.

  IF gt_kostl_filter IS NOT INITIAL.
    PERFORM filter_alv_data USING gt_kostl_filter.
  ELSE.
    MESSAGE s001 WITH '선택한 노드에 해당하는 배부 데이터가 없습니다.' DISPLAY LIKE 'W'.
  ENDIF.

*&---------------------------------------------------------------------*
*& Form check_allocation_error
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
FORM check_allocation_error  CHANGING  pt_result       LIKE gt_result
                                       pt_result_error LIKE gt_result
                                       pv_error        LIKE gv_error.

  DATA: BEGIN OF ls_seg_check,
          cycle        LIKE gs_result-cycle,
          segm_no      LIKE gs_result-segm_no,
          s_kostl      LIKE gs_result-s_kostl,
          total_skf    LIKE gs_result-total_skf,
          sum_recv_amt LIKE gs_result-recv_amt,
        END OF ls_seg_check.

  DATA: lt_seg_check LIKE HASHED TABLE OF ls_seg_check
                     WITH UNIQUE KEY cycle segm_no s_kostl,
        lv_row_error TYPE abap_bool,
        lv_err_msg   TYPE string.

  REFRESH pt_result_error.

  LOOP AT pt_result INTO DATA(ls_init).
    CLEAR ls_seg_check.
    ls_seg_check-cycle        = ls_init-cycle.
    ls_seg_check-segm_no      = ls_init-segm_no.
    ls_seg_check-s_kostl      = ls_init-s_kostl.
    ls_seg_check-total_skf    = ls_init-total_skf.
    ls_seg_check-sum_recv_amt = ls_init-recv_amt.
    COLLECT ls_seg_check INTO lt_seg_check.
  ENDLOOP.

  LOOP AT pt_result ASSIGNING FIELD-SYMBOL(<fs_res>).
    CLEAR lv_row_error.

    READ TABLE lt_seg_check INTO DATA(ls_chk)
         WITH KEY cycle   = <fs_res>-cycle
                  segm_no = <fs_res>-segm_no
                  s_kostl = <fs_res>-s_kostl.

    IF ls_chk-total_skf > 0 AND <fs_res>-skf_val = 0.
      <fs_res>-cycle = 'DELETE_NORMAL'.
      CONTINUE.
    ENDIF.

    IF <fs_res>-recv_amt = 0 AND <fs_res>-skf_val > 0.
      <fs_res>-cycle = 'DELETE_NORMAL'.
      CONTINUE.
    ENDIF.

    IF ls_chk-total_skf = 0 AND ls_chk-sum_recv_amt = 0.
      <fs_res>-cycle = 'DELETE_NORMAL'.
      CONTINUE.
    ENDIF.

    IF <fs_res>-recv_amt < 0.
      lv_row_error = abap_true.
      lv_err_msg   = |리시버 { <fs_res>-r_kostl }의 배부 금액이 음수({ <fs_res>-recv_amt })입니다.|.
    ENDIF.

    IF ls_chk-total_skf = 0 AND ls_chk-sum_recv_amt <> 0.
      lv_row_error = abap_true.
      lv_err_msg   = |센더 { <fs_res>-s_kostl }의 총 SKF가 0인데 배부액이 존재합니다.|.
    ENDIF.

    IF <fs_res>-skf_perc > 0 AND <fs_res>-skf_val = 0.
      lv_row_error = abap_true.
      lv_err_msg   = |리시버 { <fs_res>-r_kostl }의 배부 비율이 있으나 SKF 수량이 0입니다.|.
    ENDIF.

    IF lv_row_error = abap_true.
      APPEND <fs_res> TO pt_result_error.
    ENDIF.
  ENDLOOP.

  IF pt_result_error IS NOT INITIAL.
    pv_error = abap_true.
  ENDIF.

  IF pv_error IS NOT INITIAL.
    gv_error = 'X'.

    LOOP AT pt_result_error INTO DATA(ls_err).
      DELETE pt_result WHERE cycle   = ls_err-cycle
                        AND segm_no = ls_err-segm_no
                        AND s_kostl = ls_err-s_kostl
                        AND r_kostl = ls_err-r_kostl.
    ENDLOOP.

    IF gv_exit_program IS NOT INITIAL.
      IF lv_err_msg IS INITIAL.
        lv_err_msg = '배부 도중 정합성 오류가 발생했습니다.'.
      ENDIF.
      MESSAGE s001 WITH lv_err_msg DISPLAY LIKE 'I'.
      CLEAR gv_error.
      EXIT.
    ENDIF.
  ENDIF.

  DELETE pt_result WHERE cycle = 'DELETE_NORMAL'.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form dipslay_journal_temp
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM display_journal_temp.

  DATA: lv_docln   TYPE zte4fi0003-docln,
        lv_doc_seq TYPE i VALUE 0.

  DATA: BEGIN OF ls_bel_map,
          s_kostl          TYPE zee4_co_kostl,
          s_kostl_txt      TYPE zee4_co_ktext,
          s_hkont_low      TYPE zte4co0013-s_hkont_low,
          s_hkont_low_txt  TYPE zte4fi0001-txt20,
          s_hkont_high     TYPE zte4co0013-s_hkont_high,
          s_hkont_high_txt TYPE zte4fi0001-txt20,
          belnr            TYPE zte4fi0002-belnr,
        END OF ls_bel_map,
        lt_bel_map LIKE TABLE OF ls_bel_map.

  DATA: BEGIN OF ls_line_ctr,
          belnr TYPE zte4fi0002-belnr,
          ctr   TYPE i,
        END OF ls_line_ctr,
        lt_line_ctrs LIKE TABLE OF ls_line_ctr.

  DATA: lt_debit_agg LIKE TABLE OF gs_journal,
        ls_journal   LIKE gs_journal.

  REFRESH: gt_journal, lt_bel_map, lt_line_ctrs, lt_debit_agg.

  " 계정 마스터 테이블 조인 및 일괄 조회
  IF gt_result IS NOT INITIAL.
    SELECT c13~cycle,
           c13~segm_no,
           c13~s_hkont_low,
           c13~s_hkont_high,
           f1~txt20          AS s_hkont_low_txt,
           f2~txt20          AS s_hkont_high_txt
      FROM zte4co0013        AS c13
      LEFT OUTER JOIN zte4fi0001 AS f1 ON c13~s_hkont_low  = f1~saknr
      LEFT OUTER JOIN zte4fi0001 AS f2 ON c13~s_hkont_high = f2~saknr
       FOR ALL ENTRIES IN @gt_result
     WHERE c13~cycle   = @gt_result-cycle
       AND c13~segm_no = @gt_result-segm_no
      INTO TABLE @DATA(lt_orig_segments).

    SORT lt_orig_segments BY cycle segm_no.
  ENDIF.

  " 송신처 + 계정범위 단위로 전표번호 매핑
  LOOP AT gt_result INTO DATA(ls_res_init).
    READ TABLE lt_orig_segments INTO DATA(ls_orig)
         WITH KEY cycle   = ls_res_init-cycle
                  segm_no = ls_res_init-segm_no
         BINARY SEARCH.

    IF sy-subrc = 0.
      DATA(lv_orig_low)      = ls_orig-s_hkont_low.
      DATA(lv_orig_high)     = ls_orig-s_hkont_high.
      DATA(lv_orig_low_txt)  = ls_orig-s_hkont_low_txt.
      DATA(lv_orig_high_txt) = ls_orig-s_hkont_high_txt.
    ELSE.
      lv_orig_low      = ls_res_init-s_hkont_low.
      lv_orig_high     = ls_res_init-s_hkont_high.
      lv_orig_low_txt  = ls_res_init-s_hkont_low_txt.
      lv_orig_high_txt = ls_res_init-s_hkont_high_txt.
    ENDIF.

    READ TABLE lt_bel_map WITH KEY s_kostl      = ls_res_init-s_kostl
                                   s_hkont_low  = lv_orig_low
                                   s_hkont_high = lv_orig_high TRANSPORTING NO FIELDS.
    IF sy-subrc NE 0.
      lv_doc_seq = lv_doc_seq + 1.
      APPEND VALUE #( s_kostl          = ls_res_init-s_kostl
                      s_kostl_txt      = ls_res_init-s_kostl_txt
                      s_hkont_low      = lv_orig_low
                      s_hkont_low_txt  = lv_orig_low_txt
                      s_hkont_high     = lv_orig_high
                      s_hkont_high_txt = lv_orig_high_txt
                      belnr            = |TEMP{ lv_doc_seq WIDTH = 6 ALIGN = RIGHT PAD = '0' }| ) TO lt_bel_map.
    ENDIF.
  ENDLOOP.

  " 차변(수신) 데이터 사전 집계 수집
  LOOP AT gt_result INTO DATA(ls_res).
    READ TABLE lt_orig_segments INTO DATA(ls_orig_ch)
         WITH KEY cycle   = ls_res-cycle
                  segm_no = ls_res-segm_no
         BINARY SEARCH.

    IF sy-subrc = 0.
      DATA(lv_low_chk)   = ls_orig_ch-s_hkont_low.
      DATA(lv_high_chk)  = ls_orig_ch-s_hkont_high.
    ELSE.
      lv_low_chk  = ls_res-s_hkont_low.
      lv_high_chk = ls_res-s_hkont_high.
    ENDIF.

    READ TABLE lt_bel_map INTO DATA(ls_map) WITH KEY s_kostl      = ls_res-s_kostl
                                                     s_hkont_low  = lv_low_chk
                                                     s_hkont_high = lv_high_chk.

    CLEAR ls_journal.
    ls_journal-belnr       = ls_map-belnr.
    ls_journal-shkzg       = '차변(수신)'.
    ls_journal-kstar       = ls_res-kstar.
    ls_journal-kstar_txt   = ls_res-kstar_txt.
    ls_journal-debit_amt   = ls_res-recv_amt.
    ls_journal-waers       = ls_res-waers.
    ls_journal-s_kostl     = ls_res-s_kostl.
    ls_journal-s_kostl_txt = ls_res-s_kostl_txt.
    ls_journal-r_kostl     = ls_res-r_kostl.
    ls_journal-r_kostl_txt = ls_res-r_kostl_txt.
    ls_journal-r_prctr     = ls_res-r_prctr.
    ls_journal-r_prctr_txt = ls_res-r_prctr_txt.

    COLLECT ls_journal INTO lt_debit_agg.
  ENDLOOP.

  " 집계 완료된 차변(수신) 라인에 전표 세부 번호 부여 및 반영
  LOOP AT lt_debit_agg INTO ls_journal.
    READ TABLE lt_line_ctrs ASSIGNING FIELD-SYMBOL(<fs_ctr>) WITH KEY belnr = ls_journal-belnr.
    IF sy-subrc <> 0.
      APPEND VALUE #( belnr = ls_journal-belnr ctr = 0 ) TO lt_line_ctrs.
      READ TABLE lt_line_ctrs ASSIGNING <fs_ctr> WITH KEY belnr = ls_journal-belnr.
    ENDIF.
    <fs_ctr>-ctr = <fs_ctr>-ctr + 1.
    ls_journal-docln = |{ <fs_ctr>-ctr WIDTH = 6 ALIGN = RIGHT PAD = '0' }|.
    APPEND ls_journal TO gt_journal.
  ENDLOOP.

  " 대변(송신) 계정 범위 내의 계정 생성
  LOOP AT lt_bel_map INTO ls_map.
    LOOP AT gt_fi_base_costs INTO DATA(ls_base)
         WHERE s_kostl = ls_map-s_kostl
           AND racct  >= ls_map-s_hkont_low
           AND racct  <= ls_map-s_hkont_high.

      IF ls_base-total_wsl <> 0.
        READ TABLE lt_line_ctrs ASSIGNING <fs_ctr> WITH KEY belnr = ls_map-belnr.
        <fs_ctr>-ctr = <fs_ctr>-ctr + 1.
        lv_docln = |{ <fs_ctr>-ctr WIDTH = 6 ALIGN = RIGHT PAD = '0' }|.

        APPEND VALUE #(
          belnr       = ls_map-belnr
          docln       = lv_docln
          shkzg       = '대변(송신)'
          kstar       = ls_base-racct
          kstar_txt   = ls_base-racct_txt
          s_kostl     = ls_map-s_kostl
          s_kostl_txt = ls_map-s_kostl_txt
          waers       = 'KRW'
          credit_amt  = ls_base-total_wsl
        ) TO gt_journal.
      ENDIF.
    ENDLOOP.
  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form set_document_result_2
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GO_DOCUMENT_RESULT_2
*&---------------------------------------------------------------------*
FORM set_document_result_2  USING po_document TYPE REF TO cl_dd_document.

  DATA: lo_table      TYPE REF TO cl_dd_table_element,
        lo_col1       TYPE REF TO cl_dd_area,
        lo_col2       TYPE REF TO cl_dd_area,
        lv_amt_char   TYPE c LENGTH 30,
        lv_amt_string TYPE string,
        lv_cycle_text TYPE string.

  " 데이터 실시간 집계 (전체 금액, 리시버 수, CC/PC 배부 건수)
  SELECT SUM( recv_amt ) AS total_amt,
         COUNT( * )      AS receiver_cnt,
         SUM( CASE WHEN r_kostl <> @space THEN 1 ELSE 0 END ) AS cc_cnt,
         SUM( CASE WHEN r_kostl = @space AND r_prctr <> @space THEN 1 ELSE 0 END ) AS pc_cnt
    FROM @gt_result AS mock
    INTO @DATA(ls_agg).

  " 송신 코스트센터 개수 집계
  SELECT COUNT( DISTINCT s_kostl )
    FROM @gt_result AS mock
    INTO @DATA(lv_sender_cnt).

*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM save_allocation_extract.

  DATA: lt_insert  TYPE STANDARD TABLE OF zte4co0020,
        ls_insert  TYPE zte4co0020,
        lv_item_no TYPE zee4_co_item_no.

  IF gt_result IS INITIAL.
    MESSAGE '저장할 배부 결과 데이터가 없습니다.' TYPE 'S' DISPLAY LIKE 'W'.
    RETURN.
  ENDIF.

  CLEAR: lt_insert.

  LOOP AT gt_result INTO gs_result.
    CLEAR: ls_insert.
    lv_item_no = lv_item_no + 1.

    ls_insert-extract_id = gv_extract_id.
    ls_insert-item_no    = lv_item_no.

    " 배부 결과 구조체 데이터 복사
    MOVE-CORRESPONDING gs_result TO ls_insert.

    zcl_e4_cm_time=>set_timestamp(
      CHANGING
        cs_record = ls_insert ).

    APPEND ls_insert TO lt_insert.
  ENDLOOP.

  IF lt_insert IS NOT INITIAL.

    INSERT zte4co0020 FROM TABLE lt_insert.

    IF sy-subrc = 0.

      COMMIT WORK.
      MESSAGE s025 WITH '배부 추출 결과'.

    ELSE.

      ROLLBACK WORK.
      MESSAGE s026 WITH '배부 추출 결과'.

    ENDIF.
  ENDIF.

ENDFORM.
