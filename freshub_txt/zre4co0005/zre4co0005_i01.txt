MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'LIST'.
      CLEAR gv_okcode.
      CALL SCREEN '0300' STARTING AT 20 3.

    WHEN 'SEGMENT'.
      CLEAR gv_okcode.
      PERFORM get_segment_data.
      PERFORM refresh_segment.

    WHEN 'SETTING'.
      CLEAR gv_okcode.
      CALL SCREEN '0400' STARTING AT 20 3.

    WHEN 'EXECUTE'.
      CLEAR gv_okcode.
      PERFORM get_selected_cycle.
*      PERFORM check_background_job.
      PERFORM calculate_allocation_data.
      PERFORM handle_execution_mode.

  ENDCASE.

ENDMODULE.
