*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_plan .

  DATA : lv_last_date  TYPE d, " 계획 연월 검색
         lv_first_date TYPE d,
         lr_date       TYPE RANGE OF zte2pp0007-pdatu,
         ls_date       LIKE LINE OF lr_date,
         lr_stat       TYPE RANGE OF zte2pp0007-versp,
         ls_stat       LIKE LINE OF lr_stat.

  CLEAR   : gs_plan.
  REFRESH : gt_plan.

  IF gs_search-spmon IS NOT INITIAL.

    lv_first_date = gs_search-spmon && '01'.

    CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
      EXPORTING
        day_in            = lv_first_date
      IMPORTING
        last_day_of_month = lv_last_date.

    ls_date-low    = lv_first_date.
    ls_date-high   = lv_last_date.
    ls_date-sign   = 'I'.
    ls_date-option = 'BT'.

    APPEND ls_date TO lr_date.
    CLEAR  ls_date.

  ENDIF.

  IF gs_search-sel_uncon IS NOT INITIAL.
    ls_stat-sign   = 'I'.
    ls_stat-option = 'EQ'.
    ls_stat-low    = ' '. " 공백인 놈들만 가져와라

    APPEND ls_stat TO lr_stat.
    CLEAR  ls_stat.

  ENDIF.

*  SELECT a~plnum b~matnr a~pernr
*         b~posnr b~pdatu b~plnmg b~meins b~versp
*         c~maktx
*         d~sname
*    INTO CORRESPONDING FIELDS OF TABLE gt_plan
*    FROM zte2pp0006 AS a
*   INNER JOIN zte2pp0007 AS b
*      ON a~plnum = b~plnum
*   INNER JOIN zte2mm0001 AS c
*      ON b~matnr = c~matnr
*   INNER JOIN zte2hr0002 AS d
*      ON a~pernr = d~pernr
*   WHERE b~pdatu IN lr_date
*     AND b~versp IN lr_stat.

  SELECT a~plnum, b~matnr, a~pernr,
         b~posnr, b~pdatu, b~plnmg, b~meins, b~versp,
         c~maktx,
         d~sname,
         CASE b~versp
         WHEN 'X' THEN '오더 전환 완료'
         ELSE '오더 전환 대기' END AS stat_txt
    FROM zte2pp0006 AS a
   INNER JOIN zte2pp0007 AS b
      ON a~plnum = b~plnum
   INNER JOIN zte2mm0001 AS c
      ON b~matnr = c~matnr
   INNER JOIN zte2hr0002 AS d
      ON a~pernr = d~pernr
   WHERE b~pdatu IN @lr_date
     AND b~versp IN @lr_stat
   ORDER BY b~pdatu
    INTO CORRESPONDING FIELDS OF TABLE @gt_plan.

IF sy-subrc <> 0.
  IF gv_ok = 'X'.
    MESSAGE s000 WITH '생산계획이 생산오더로 모두 전환되었습니다.'.
    CLEAR gv_ok.
  ELSE.
    MESSAGE s000 WITH '해당 조건에 일치하는 생산계획이 없습니다.' DISPLAY LIKE 'E'.
    RETURN.
  ENDIF.
  EXIT.
ENDIF.

  DATA : lv_data TYPE i.

  SELECT COUNT(*)
    INTO lv_data
    FROM zte2pp0007
   WHERE pdatu IN lr_date
     AND versp = 'X'.

  gs_doc-converted = |{ lv_data }|.

  SELECT COUNT(*)
    INTO lv_data
    FROM zte2pp0007
   WHERE pdatu IN lr_date
     AND versp = ' '.

  gs_doc-pending = |{ lv_data }|.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING    ps_row TYPE lvc_s_row
                              ps_col TYPE lvc_s_col.

  READ TABLE gt_plan INTO gs_plan INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.
      PERFORM get_infos USING gs_plan-matnr.
      PERFORM get_bom USING gs_plan-matnr.
      PERFORM get_rou USING gs_plan-matnr.
      PERFORM refresh_bom.
      PERFORM refresh_rou.

  ENDCASE.

ENDFORM.

  CLEAR gs_info.

  SELECT SINGLE stlnr
    INTO CORRESPONDING FIELDS OF gs_info
    FROM zte2pp0008
   WHERE matnr = pv_matnr.

    INTO CORRESPONDING FIELDS OF gs_info
    FROM zte2pp0008 AS a
   INNER JOIN zte2pp0004 AS b
      ON a~plnnr = b~plnnr
   WHERE a~matnr = pv_matnr.

  CALL METHOD cl_gui_cfw=>set_new_ok_code
    EXPORTING
      new_code = 'GO_PAI'.

ENDFORM.

  ENDLOOP.

*&---------------------------------------------------------------------*
*& Form create_order
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM create_order .

  DATA : ls_save_h TYPE zte2pp0009,
         ls_save_i TYPE zte2pp0010,
         lt_save_h LIKE TABLE OF ls_save_h,
         lt_save_i LIKE TABLE OF ls_save_i,
         ls_row    TYPE lvc_s_row,
         lt_row    TYPE lvc_t_row,
         lv_commit,
         lv_cnt    TYPE i,
         lv_aufnr  TYPE zte2pp0009-aufnr,
         BEGIN OF ls_detail,
           stlnr TYPE zte2pp0009-stlnr,
           plnnr TYPE zte2pp0009-plnnr,
         END OF ls_detail,
         ls_timestamp TYPE zse2timestamp,

         ls_plan      TYPE zte2pp0007,
         lt_plan      LIKE TABLE OF zte2pp0007.

  CALL METHOD gcl_grid_plan->get_selected_rows
    IMPORTING
      et_index_rows = lt_row.

  lv_cnt = lines( lt_row ).

  IF lv_cnt < 1.
    MESSAGE s000 WITH '오더로 전환할 행을 선택해주세요.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  LOOP AT lt_row INTO ls_row.

    READ TABLE gt_plan INTO gs_plan INDEX ls_row-index.

    IF sy-subrc = 0.

      IF gs_plan-versp = 'X'.
        MESSAGE s000 WITH '이미 전환된 생산계획입니다.' DISPLAY LIKE 'E'.
        RETURN.
      ENDIF.

      PERFORM make_gr_aufnr USING lv_aufnr.

      SELECT SINGLE stlnr plnnr
        INTO CORRESPONDING FIELDS OF ls_detail
        FROM zte2pp0008
       WHERE matnr = gs_plan-matnr.

      MOVE-CORRESPONDING ls_detail TO ls_save_h.

      ls_save_h = VALUE #( BASE ls_save_h
                                aufnr = lv_aufnr        " 생산오더번호
                                auart = 'ZFCT'          " 생산오더 유형
                                plnum = gs_plan-plnum   " 생산계획번호
                                plpos = gs_plan-posnr   " 생산계획항목
                                plnbez = gs_plan-matnr   " 자재번호
                                gamng = gs_plan-plnmg   " 생산 지시 수량
                                meins = gs_plan-meins   " 단위
                                gstrp = gs_plan-pdatu   " 계획 시작일
                                gltrp = gs_plan-pdatu   " 계획종료일 이거 추후수정 ㅇ
                                apstat = '1'            " 대기중
                                ordat = sy-datum        " 오더 생성일
                           ).

      MOVE-CORRESPONDING gs_plan TO ls_timestamp.

      CALL FUNCTION 'ZFE2US0002'
        EXPORTING
          is_timestamp = ls_timestamp
        IMPORTING
          es_timestamp = ls_timestamp.

      MOVE-CORRESPONDING ls_timestamp TO ls_save_h.

      APPEND ls_save_h TO lt_save_h.
      CLEAR  ls_save_h.

      " 오더전환여부 체크 - plan테이블에 ㅇㅇ
      MOVE-CORRESPONDING gs_plan TO ls_plan.
      ls_plan-versp = 'X'.
      APPEND ls_plan TO lt_plan.
      CLEAR : ls_plan, gs_plan.

    ENDIF.

  ENDLOOP.

  IF lt_save_h IS NOT INITIAL.
    MODIFY zte2pp0009 FROM TABLE lt_save_h.
    IF sy-subrc <> 0.
      lv_commit = 'X'.
    ENDIF.

    MODIFY zte2pp0007 FROM TABLE lt_plan.
    IF sy-subrc <> 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit <> 'X'.
    COMMIT WORK AND WAIT.
    gv_ok = 'X'.
    MESSAGE s000 WITH '생산오더로 전환이 완료되었습니다.'.
    PERFORM get_plan.
    PERFORM set_icon.
    CLEAR  : gs_info_bom, gs_info_rou.
    PERFORM refresh_grid.
    PERFORM refresh_screen.
    " 나중에 생산오더에없는거만 전환해야함.
    " 애초에 전환된거 versp = 'X'는 전환 못하게 !!!
  ELSE.
    ROLLBACK WORK.
    MESSAGE s000 WITH '생산오더 전환중 문제가 생겼습니다.' DISPLAY LIKE 'E'.

    CLEAR lv_commit.
  ENDIF.

ENDFORM.

*&      --> GS_PLAN_MATNR

FORM get_bom  USING    pv_matnr TYPE zte2mm0001-matnr.

  SELECT SINGLE stlnr bmeng meins
    INTO CORRESPONDING FIELDS OF gs_info_bom
    FROM zte2pp0002
   WHERE matnr = pv_matnr.

  SELECT a~idnrk a~menge a~meins
         b~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_bom
    FROM zte2pp0002 AS c
   INNER JOIN zte2pp0003 AS a
      ON a~stlnr = c~stlnr
   INNER JOIN zte2mm0001 AS b
      ON a~idnrk = b~matnr
   WHERE c~matnr = pv_matnr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '해당 생산의 BOM이 존재하지 않습니다.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_rou
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_PLAN_MATNR
*&---------------------------------------------------------------------*
FORM get_rou  USING    pv_matnr TYPE zte2mm0001-matnr.

  SELECT SINGLE a~plnnr b~ktext c~werks b~runyn a~bmsch a~meins
    INTO CORRESPONDING FIELDS OF gs_info_rou
    FROM zte2pp0005 AS  a
   INNER JOIN zte2pp0004 AS b
      ON a~plnnr = b~plnnr
   INNER JOIN zte2pp0001 AS c
      ON a~arbpl = c~arbpl
   INNER JOIN zte2pp0008 AS d
      ON d~matnr = pv_matnr
     AND d~plnnr = b~plnnr.

  CASE gs_info_rou-runyn.
    WHEN 'Y'.
      gs_info_rou-text = '가동'.
    WHEN 'N'.
      gs_info_rou-text = '정지'.
  ENDCASE.


  SELECT a~vornr a~arbpl a~ltxa1
         b~ktext
    INTO CORRESPONDING FIELDS OF TABLE gt_rou
    FROM zte2pp0005 AS a
   INNER JOIN zte2pp0001 AS b
      ON a~arbpl = b~arbpl
   INNER JOIN zte2pp0008 AS c
      ON c~matnr = pv_matnr
     AND c~plnnr = a~plnnr
   ORDER BY a~vornr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '해당 생산의 라우팅이 존재하지 않습니다.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

ENDFORM.
