*& Form get_form
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> TEXT_M02
*&      --> LV_FORM_ID
*&---------------------------------------------------------------------*
FORM get_form  USING  pv_name
                      pv_form.

  DATA : lv_filename    TYPE string,
         lv_name        TYPE string,
         lv_path        TYPE string,
         lv_fullpath    TYPE string,
         lv_user_action TYPE i,
         ls_key         TYPE wwwdatatab,
         lv_langu       TYPE laiso.

  SELECT SINGLE f~relid f~objid f~checkout f~checknew
                f~chname f~tdate f~ttime f~text p~devclass
           INTO CORRESPONDING FIELDS OF ls_key
           FROM wwwdata AS f
           JOIN tadir AS p
             ON f~objid    EQ p~obj_name
          WHERE f~srtf2    EQ 0
            AND f~relid    EQ 'MI'
            AND p~pgmid    EQ 'R3TR'
            AND p~object   EQ 'W3MI'
            AND p~obj_name EQ pv_form.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH TEXT-e01 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  CONCATENATE pv_name '생산 계획 업로드 엑셀 양식.xlsx' INTO lv_name.

  CALL METHOD cl_gui_frontend_services=>file_save_dialog
    EXPORTING
*     window_title              =
      default_extension         = 'XLSX'
      default_file_name         = lv_name
*     with_encoding             =
*     file_filter               =
*     initial_directory         =
*     prompt_on_overwrite       = 'X'
    CHANGING
      filename                  = lv_filename
      path                      = lv_path
      fullpath                  = lv_fullpath
      user_action               = lv_user_action
*     file_encoding             =
    EXCEPTIONS
      cntl_error                = 1
      error_no_gui              = 2
      not_supported_by_gui      = 3
      invalid_default_file_name = 4
      OTHERS                    = 5.

  IF lv_user_action = cl_gui_frontend_services=>action_cancel.
    EXIT.
  ENDIF.

  CALL FUNCTION 'DOWNLOAD_WEB_OBJECT'
    EXPORTING
      key         = ls_key
      destination = CONV localfile( lv_fullpath ).

  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document               = lv_fullpath
    EXCEPTIONS
      cntl_error             = 1
      error_no_gui           = 2
      bad_parameter          = 3
      file_not_found         = 4
      path_not_found         = 5
      file_extension_unknown = 6
      error_execute_failed   = 7
      synchronous_failed     = 8
      not_supported_by_gui   = 9
      OTHERS                 = 10.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form delete_plan
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM delete_plan .

  DATA : lv_cnt TYPE i,
         ls_del LIKE LINE OF gt_plan_del.


  CALL METHOD gcl_grid_plan->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  lv_cnt = lines( gt_row ).

  IF lv_cnt < 1.
    MESSAGE s000 WITH '먼저 행을 선택해주세요.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT  gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_plan INTO gs_plan INDEX gs_row-index.

    IF sy-subrc = 0.

      MOVE-CORRESPONDING gs_plan TO ls_del.

      APPEND ls_del TO gt_plan_del. " 생각해보니까이게필요가없네
      CLEAR ls_del.

      DELETE gt_plan INDEX gs_row-index.
      MESSAGE s000 WITH '삭제가 완료되었습니다.'.

    ENDIF.

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_changed_finished
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_MODIFIED
*&      --> ET_GOOD_CELLS
*&---------------------------------------------------------------------*
FORM handle_changed_finished   USING   pv_modified
                                       pt_modi TYPE lvc_t_modi.

  DATA : ls_modi TYPE lvc_s_modi,
         BEGIN OF gs_spmon,
           gjahr TYPE zte2sd0015-gjahr,
           poper TYPE zte2sd0016-poper,
         END OF gs_spmon,
         lv_maktx TYPE zte2mm0001-maktx.

  CHECK pv_modified IS NOT INITIAL.

  LOOP AT pt_modi INTO ls_modi.

    READ TABLE gt_plan INTO gs_plan INDEX ls_modi-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

*    gs_plan-mark = 'X'.

    CASE ls_modi-fieldname. " 나중에 단위필드도 막기.
      WHEN 'MATNR'.
        SELECT SINGLE maktx
          INTO gs_plan-maktx
          FROM zte2mm0001
         WHERE matnr = gs_plan-matnr.

      WHEN 'PBDNR'.
        SELECT SINGLE a~gjahr b~poper
          INTO gs_spmon
          FROM zte2sd0015 AS a
         INNER JOIN zte2sd0016 AS b
            ON a~plnnr = b~plnnr
         WHERE b~plnnr = gs_plan-pbdnr.

        IF sy-subrc = 0.
          CONCATENATE gs_spmon-gjahr gs_spmon-poper+1(2) INTO gs_plan-spmon.
        ENDIF.
    ENDCASE.

    MODIFY gt_plan FROM gs_plan INDEX ls_modi-row_id
    TRANSPORTING maktx spmon.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM create_plan .

  DATA : ls_save_h    TYPE zte2pp0006,
         ls_save_i    TYPE zte2pp0007,
         lt_save_h    LIKE TABLE OF ls_save_h,
         lt_save_i    LIKE TABLE OF ls_save_i,
         lv_commit,
         lv_plnum     TYPE zte2pp0006-plnum,
         lv_posnr     TYPE zte2pp0007-posnr,
         ls_timestamp TYPE zse2timestamp,
         lv_err.

  CLEAR   : ls_save_h, ls_save_i, lv_commit.
  REFRESH : lt_save_h, lt_save_i.

  PERFORM valid_check CHANGING lv_err.
  IF lv_err IS NOT INITIAL .
    RETURN.
  ENDIF.

  SORT gt_plan BY pbdnr.

  DATA(ls_temp) = gs_plan.

  LOOP AT gt_plan INTO gs_plan.

    CALL FUNCTION 'ZFE2US0002'
      EXPORTING
        is_timestamp = ls_timestamp
      IMPORTING
        es_timestamp = ls_timestamp.

    AT NEW pbdnr.
      CLEAR : ls_save_h, lv_posnr.
      " 하나의 헤더 기준 - 월
      PERFORM make_gr_plnum USING lv_plnum.
      ls_save_h-plnum = lv_plnum.
      ls_save_h-werks = '1000'.
      ls_save_h-spmon = pa_spmon.
      ls_save_h-pernr = gs_pernr-pernr.

      MOVE-CORRESPONDING ls_temp TO ls_save_h.
      MOVE-CORRESPONDING ls_timestamp TO ls_save_h.
      APPEND ls_save_h TO lt_save_h.

    ENDAT.
    lv_posnr += 1. " 루프 돌때마다 +10 (한줄당 10)

    ls_save_i-plnum = lv_plnum.
    ls_save_i-posnr = lv_posnr.

    MOVE-CORRESPONDING gs_plan TO ls_save_i.
    MOVE-CORRESPONDING ls_timestamp TO ls_save_i.
    APPEND ls_save_i TO lt_save_i.

  ENDLOOP.

  IF lt_save_h IS NOT INITIAL.
    MODIFY zte2pp0006 FROM TABLE lt_save_h.
    MODIFY zte2pp0007 FROM TABLE lt_save_i. " 추후 변경

    IF sy-subrc = 0.
      COMMIT WORK.
      MESSAGE s000 WITH '생산계획이 생성되었습니다.'.
**********************************************************************
* 생산계획 찐 생성시 Refresh 를해줘야하나 ㅏㅏㅏ  ..
**********************************************************************
      REFRESH gt_plan.
      PERFORM refresh_grid.
      CLEAR   pa_fname.
      CLEAR   gs_pernr.

    ELSE.
      ROLLBACK WORK.
      MESSAGE s000 WITH '생산계획 생성 중 오류가 발생했습니다.' DISPLAY LIKE 'E'.
    ENDIF.

  ENDIF.

ENDFORM.

FORM f4_pernr .

  DATA: BEGIN OF ls_pernr,
          pernr TYPE zte2hr0002-pernr,
          sname TYPE zte2hr0002-sname, " 사원 성명
          orgtx TYPE zte2hr0001-orgtx, " 부서명
        END OF ls_pernr,

        lt_pernr  LIKE TABLE OF ls_pernr,

        lt_return TYPE TABLE OF ddshretval,
        ls_return TYPE ddshretval,
        lt_dynp   TYPE TABLE OF dynpread,
        ls_dynp   TYPE dynpread.

  SELECT a~pernr, a~sname, b~orgtx
    FROM zte2hr0002 AS a
   INNER JOIN zte2hr0001 AS b
      ON a~orgeh = b~orgeh
     AND a~orgeh = '30000000'
     INTO CORRESPONDING FIELDS OF TABLE @lt_pernr.


  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'PERNR'
      dynpprog     = sy-repid
      dynpnr       = sy-dynnr
      dynprofield  = 'GS_PERNR-PERNR'
      window_title = '담당자 지정'
      value_org    = 'S'
    TABLES
      value_tab    = lt_pernr
      return_tab   = lt_return.

  IF lt_return[] IS NOT INITIAL.

    READ TABLE lt_return INTO ls_return INDEX 1.

    IF sy-subrc = 0.

      READ TABLE lt_pernr INTO ls_pernr WITH KEY pernr = ls_return-fieldval.

      IF sy-subrc = 0.
        CLEAR   ls_dynp.

        ls_dynp-fieldname = 'GS_PERNR-PERNR'.
        ls_dynp-fieldvalue = ls_return-fieldval.
        APPEND ls_dynp TO lt_dynp.
        CLEAR ls_dynp.

        ls_dynp-fieldname = 'GS_PERNR-SNAME'.
        ls_dynp-fieldvalue = ls_pernr-sname.
        APPEND ls_dynp TO lt_dynp.
        CLEAR ls_dynp.

        CALL FUNCTION 'DYNP_VALUES_UPDATE'
          EXPORTING
            dyname               = sy-repid
            dynumb               = sy-dynnr
          TABLES
            dynpfields           = lt_dynp
          EXCEPTIONS
            invalid_abapworkarea = 1
            invalid_dynprofield  = 2
            invalid_dynproname   = 3
            invalid_dynpronummer = 4
            invalid_request      = 5
            no_fielddescription  = 6
            undefind_error       = 7
            OTHERS               = 8.

      ENDIF.

      gs_pernr-pernr = ls_pernr-pernr.
      gs_pernr-sname = ls_pernr-sname.

    ENDIF.

  ENDIF.


ENDFORM.
