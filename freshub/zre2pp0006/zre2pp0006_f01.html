FORM delete_plan .

    IMPORTING

    ENDIF.

  ENDLOOP.

FORM create_plan .

  DATA : ls_save_h    TYPE zte2pp0006,

  REFRESH : lt_save_h, lt_save_i.

  PERFORM valid_check CHANGING lv_err.

    RETURN.
  ENDIF.

  SORT gt_plan BY pbdnr.

    ENDAT.
    lv_posnr += 1. " 루프 돌때마다 +10 (한줄당 10)

    ls_save_i-plnum = lv_plnum.
    ls_save_i-posnr = lv_posnr.

    MOVE-CORRESPONDING ls_timestamp TO ls_save_i.
    APPEND ls_save_i TO lt_save_i.

  ENDLOOP.

  IF lt_save_h IS NOT INITIAL.

      REFRESH gt_plan.
