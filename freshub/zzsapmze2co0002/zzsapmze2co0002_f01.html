FORM get_materials .

  SELECT a~matnr, a~maktx, b~werks, c~name1,
         d~bdatj, d~poper,
        CASE
         WHEN d~matnr IS NULL THEN '@5C@' " 데이터 없으면
         ELSE '@5B@'                      " 데이터 있으면
       END AS status
    FROM zte2mm0001 AS a      " 자재마스터 -> matnr, maktx
   INNER JOIN zte2pp0008 AS b " 생산품 - 라우팅 - BOM  연결
      ON a~matnr = b~matnr
   INNER JOIN zte2mm0002 AS c "플랜터마스터 -> name1
      ON b~werks = c~werks
    LEFT OUTER JOIN zte2co0005 AS d " 표준원가 -> 상태
      ON a~matnr = d~matnr
     AND d~bdatj = @gs_search-bdatj
     AND d~poper = @gs_search-poper
   ORDER BY CASE
    WHEN d~matnr IS NULL THEN 0
    ELSE 1
     END,
    a~matnr
  INTO CORRESPONDING FIELDS OF TABLE @gt_matnr.

ENDFORM.

FORM get_cost .

  CLEAR   gs_cost.
  REFRESH gt_cost.

  SELECT a~bukrs, a~matnr, a~werks, a~bdatj, a~poper, a~vadat,
         a~peinh, a~meins, a~dmatl, a~dlabr, a~mohad, a~stprs, a~waers,
         b~maktx
    FROM zte2co0005 AS a
   INNER JOIN zte2mm0001 AS b
      ON a~matnr = b~matnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_cost.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH '해당 년도의 표준원가가 없습니다.' DISPLAY LIKE 'W'.
    EXIT.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_hierarchy .

  DATA : lv_first.

  DATA : BEGIN OF ls_year,
           bdatj TYPE zte2co0005-bdatj,
         END OF ls_year,

         BEGIN OF ls_month,
           bdatj TYPE zte2co0005-bdatj,
           poper TYPE zte2co0005-poper,
         END OF ls_month,

         lt_year          LIKE TABLE OF ls_year,
         lt_month         LIKE TABLE OF ls_month,

         lt_data          LIKE TABLE OF gs_cost,
         ls_empty         LIKE gs_cost,

         lv_level_1       TYPE lvc_nkey,
         lv_level_2       TYPE lvc_nkey,
         lv_node_key      TYPE lvc_nkey,
         lv_tabix         TYPE sy-tabix,

         lv_year_text(6),
         lv_month_text(5),

* 검색 itab 구성키 위한 변수들
         lv_key_1         TYPE tv_nodekey,
         lv_key_2         TYPE tv_nodekey,
         lv_cnt           TYPE sy-tabix.

  CLEAR   : ls_year, ls_month, lv_first, ls_empty, lv_key_1, lv_key_2, lv_cnt, lv_first.
  REFRESH : lt_data, lt_year, lt_month.

  APPEND LINES OF gt_cost TO lt_data.

  REFRESH gt_cost.

  SORT lt_data BY bdatj poper matnr.

  lt_year  = CORRESPONDING #( lt_data ).
  lt_month = CORRESPONDING #( lt_data ).

  SORT : lt_year  BY bdatj,
         lt_month BY bdatj poper.

  DELETE ADJACENT DUPLICATES FROM : lt_year  COMPARING bdatj,
                                    lt_month COMPARING bdatj poper.

  MOVE-CORRESPONDING lt_year  TO gt_year.
  MOVE-CORRESPONDING lt_month TO gt_month.

  LOOP AT lt_year INTO ls_year.

    READ TABLE lt_year INTO ls_year
         WITH KEY bdatj = ls_year-bdatj
         BINARY SEARCH.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    lv_tabix = sy-tabix.
    lv_cnt  += 1.
    lv_key_1 = lv_cnt.

    READ TABLE lt_data INTO gs_cost
         WITH KEY bdatj = ls_year-bdatj
         BINARY SEARCH.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    " Level 1 Node
    lv_year_text = | { gs_cost-bdatj }년 |.

    PERFORM add_node USING ls_empty lv_year_text ' ' lv_level_1.

    LOOP AT lt_month INTO ls_month FROM lv_tabix WHERE bdatj = ls_year-bdatj.

      READ TABLE lt_data INTO gs_cost WITH KEY bdatj = ls_month-bdatj
                                               poper = ls_month-poper
                                               BINARY SEARCH.

      IF sy-subrc <> 0.
        CONTINUE.
      ENDIF.

      lv_tabix = sy-tabix.

      CLEAR lv_first.

      LOOP AT lt_data INTO gs_cost FROM lv_tabix WHERE bdatj = ls_month-bdatj
                                                   AND poper = ls_month-poper.

        IF lv_first IS INITIAL.

          lv_month_text = |{ abs( gs_cost-poper ) }월|.

          " Level 2 Node
          PERFORM add_node USING ls_empty lv_month_text lv_level_1 lv_level_2.

          lv_first = 'X'.
          lv_cnt  += 1.

  DATA : lv_tabix TYPE sy-tabix,
         lv_price LIKE gs_labor_cost-price.

  REFRESH gt_labor_cost.

  SELECT vornr, arbpl, arbpltxt, opertext, stdtime, timeunit, baseqty, meins,
         lstar1, lstar2, lstar3, price1, price2, price3, waers
    FROM zcds_e2_co_0002( p_matnr = @pv_matnr,
                          p_gjahr = @gs_search-bdatj,
                          p_perio = @gs_search-poper
                         )
    INTO CORRESPONDING FIELDS OF TABLE @gt_labor_cost
   ORDER BY vornr.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH '노무비 불러오기 오류' DISPLAY LIKE 'E'.
  ENDIF.

  LOOP AT gt_labor_cost INTO gs_labor_cost.

    lv_tabix = sy-tabix.

    IF gs_labor_cost-lstar1 = 'LABR'.
      lv_price = gs_labor_cost-price1.
    ENDIF.

    IF gs_labor_cost-lstar2 = 'LABR'.
      lv_price = gs_labor_cost-price2.
    ENDIF.

    IF gs_labor_cost-lstar3 = 'LABR'.
      lv_price = gs_labor_cost-price3.
    ENDIF.

    " gs_labor_cost-stdtime " 시간 min  : 60 Min =  price : lv_price
    " --> 60 * price  = lv_price * stdtime
    " --> price = lv_price * stdtime / 60 -> 100개당 가격
    lv_price = ( ( lv_price * gs_labor_cost-stdtime ) / 60 ) / 100.

    gs_labor_cost-price = lv_price.

    MODIFY gt_labor_cost FROM gs_labor_cost INDEX lv_tabix
    TRANSPORTING price.

  ENDLOOP.

ENDFORM.

FORM get_indirect_cost USING pv_matnr TYPE zte2mm0001-matnr.

  DATA: lv_last_day   TYPE sy-datum,
        lv_day_in     TYPE sy-datum, " 이번달 며칠인지
        lv_days       TYPE i,
        lv_1hour_quan TYPE zte2co0006-wkg00, " 1시간(60분)당 간접비 얼마인지
        lv_mach       TYPE zte2co0005-stprs,
        lv_prod       TYPE zte2co0005-stprs,
        lv_watr       TYPE zte2co0005-stprs,
        lv_elec       TYPE zte2co0005-stprs,
        lv_ict        TYPE zte2co0005-stprs,

        lv_beg_sec    TYPE i,
        lv_end_sec    TYPE i,
        lv_work_sec   TYPE i,
        lv_work_hour  TYPE p DECIMALS 2,

        lt_cycle      LIKE TABLE OF gs_indirect_cost.

  CLEAR : lv_last_day, lv_day_in, lv_days, lv_1hour_quan, lv_mach, lv_prod, lv_watr, lv_elec.

  SELECT arbpl, begzt, endzt, kostl, plnnr, vornr, vgw02, vge02, bmsch,
         meins_quan, belnr, buzei, gjahr, perio, wkg00, waers, ref_bn
  FROM zcds_e2_co_0006( p_matnr = @pv_matnr,
                          p_gjahr = @gs_search-bdatj,
                          p_perio = @gs_search-poper
                         )
    INTO CORRESPONDING FIELDS OF TABLE @gt_ind_tmp
   ORDER BY kostl, vornr.

  IF sy-subrc <> 0.
    MESSAGE s000 WITH '간접비 불러오기 오류'.
  ENDIF.

**********************************************************************
* 간접비 종류 별 가격 측정
**********************************************************************
  lv_day_in = |{ gs_search-bdatj }{ gs_search-poper+1(2) }01|.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      day_in            = lv_day_in
    IMPORTING
      last_day_of_month = lv_last_day.

  lv_days = lv_last_day+6(2).

  LOOP AT gt_ind_tmp INTO gs_ind_tmp.

    " 가동 시간 계산
    lv_beg_sec =
    gs_ind_tmp-begzt+0(2) * 3600
  + gs_ind_tmp-begzt+2(2) * 60
  + gs_ind_tmp-begzt+4(2).

    lv_end_sec =
        gs_ind_tmp-endzt+0(2) * 3600
      + gs_ind_tmp-endzt+2(2) * 60
      + gs_ind_tmp-endzt+4(2).

    lv_work_sec = lv_end_sec - lv_beg_sec.

    lv_work_hour = lv_work_sec / 3600.

    " 한달 총 간접비 / ( 가동 종료시간 - 시작시간 * 한달일수 ) => 시간당 간접비
    lv_1hour_quan = gs_ind_tmp-wkg00 / ( lv_work_hour * lv_days ).

    CASE gs_ind_tmp-ref_bn.

      WHEN '26_A_MACH'.
        "  ( 시간당 간접비 / 60 -> 분당 간접비 ) * 해당 공정 분수 / 개수
        lv_mach += ( ( gs_ind_tmp-vgw02 * lv_1hour_quan / 60 ) / gs_ind_tmp-bmsch ).

      WHEN '26_D_PROD'.
        lv_prod += ( ( gs_ind_tmp-vgw02 * lv_1hour_quan / 60 ) / gs_ind_tmp-bmsch ).

      WHEN '26_A_WATR'.
        lv_watr += ( ( gs_ind_tmp-vgw02 * lv_1hour_quan / 60 ) / gs_ind_tmp-bmsch ).

      WHEN '26_A_ELEC'.
        lv_elec += ( ( gs_ind_tmp-vgw02 * lv_1hour_quan / 60 ) / gs_ind_tmp-bmsch ).

      WHEN '26_D_ICT'.
        lv_ict += ( ( gs_ind_tmp-vgw02 * lv_1hour_quan / 60 ) / gs_ind_tmp-bmsch ).

    ENDCASE.

  ENDLOOP.

**********************************************************************
* 화면에 간접비 추가
**********************************************************************

  lt_cycle = VALUE #(
    ( ref_bn = '26_A_MACH' wkg00 = lv_mach )
    ( ref_bn = '26_A_ELEC' wkg00 = lv_elec )
    ( ref_bn = '26_A_WATR' wkg00 = lv_watr )
    ( ref_bn = '26_D_PROD' wkg00 = lv_prod )
    ( ref_bn = '26_D_ICT'  wkg00 = lv_ict  )
  ).

  LOOP AT lt_cycle INTO DATA(ls_cycle).

    gs_indirect_cost-ref_bn = ls_cycle-ref_bn.

    SELECT SINGLE text
      FROM zte2co0013
     WHERE cycle = @ls_cycle-ref_bn
      INTO @gs_indirect_cost-ref_txt.

    gs_indirect_cost-wkg00 = ls_cycle-wkg00.
    gs_indirect_cost-waers = gs_ind_tmp-waers.

    APPEND gs_indirect_cost TO gt_indirect_cost.
    CLEAR gs_indirect_cost.

  ENDLOOP.

ENDFORM.
