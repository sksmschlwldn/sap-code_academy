FORM handle_hotspot USING ps_row TYPE lvc_s_row
                              ps_col TYPE lvc_s_col.

  READ TABLE gt_plan INTO gs_plan INDEX ps_row-index.

  IF sy-subrc <> 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.
      PERFORM get_infos USING gs_plan-matnr.
      PERFORM get_bom USING gs_plan-matnr.
      PERFORM get_rou USING gs_plan-matnr.
      PERFORM refresh_bom.
      PERFORM refresh_rou.

  ENDCASE.

      ON a~plnnr = b~plnnr

FORM set_icon .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_plan INTO gs_plan.

    lv_tabix = sy-tabix.

    CASE gs_plan-versp.
      WHEN 'X'.
        gs_plan-status_icon = icon_led_green.
      WHEN OTHERS.
        gs_plan-status_icon = icon_led_yellow.
    ENDCASE.


    MODIFY gt_plan FROM gs_plan INDEX lv_tabix
    TRANSPORTING status_icon .

  ENDLOOP.

ENDFORM.

*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM create_order .

  DATA : ls_save_h TYPE zte2pp0009,
         ls_save_i TYPE zte2pp0010,
         lt_save_h LIKE TABLE OF ls_save_h,
         lt_save_i LIKE TABLE OF ls_save_i,
         ls_row    TYPE lvc_s_row,
         lt_row    TYPE lvc_t_row,
         lv_commit,
         lv_cnt    TYPE i,
         lv_aufnr  TYPE zte2pp0009-aufnr,
         BEGIN OF ls_detail,
           stlnr TYPE zte2pp0009-stlnr,
           plnnr TYPE zte2pp0009-plnnr,
         END OF ls_detail,
         ls_timestamp TYPE zse2timestamp,

         ls_plan      TYPE zte2pp0007,
         lt_plan      LIKE TABLE OF zte2pp0007.

  CALL METHOD gcl_grid_plan->get_selected_rows
    IMPORTING
      et_index_rows = lt_row.

  lv_cnt = lines( lt_row ).

  IF lv_cnt < 1.
    MESSAGE s000 WITH '오더로 전환할 행을 선택해주세요.' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  LOOP AT lt_row INTO ls_row.

    READ TABLE gt_plan INTO gs_plan INDEX ls_row-index.

    IF sy-subrc = 0.

      IF gs_plan-versp = 'X'.
        MESSAGE s000 WITH '이미 전환된 생산계획입니다.' DISPLAY LIKE 'E'.
        RETURN.
      ENDIF.

      PERFORM make_gr_aufnr USING lv_aufnr.

      SELECT SINGLE stlnr plnnr
        INTO CORRESPONDING FIELDS OF ls_detail
        FROM zte2pp0008
       WHERE matnr = gs_plan-matnr.

      MOVE-CORRESPONDING ls_detail TO ls_save_h.

      ls_save_h = VALUE #( BASE ls_save_h
                                aufnr = lv_aufnr        " 생산오더번호
                                auart = 'ZFCT'          " 생산오더 유형
                                plnum = gs_plan-plnum   " 생산계획번호
                                plpos = gs_plan-posnr   " 생산계획항목
                                plnbez = gs_plan-matnr   " 자재번호
                                gamng = gs_plan-plnmg   " 생산 지시 수량
                                meins = gs_plan-meins   " 단위
                                gstrp = gs_plan-pdatu   " 계획 시작일
                                gltrp = gs_plan-pdatu   " 계획종료일 이거 추후수정 ㅇ
                                apstat = '1'            " 대기중
                                ordat = sy-datum        " 오더 생성일
                           ).

      MOVE-CORRESPONDING gs_plan TO ls_timestamp.

      call function 'ZFE2US0002'
        EXPORTING
          is_timestamp = ls_timestamp
        IMPORTING
          es_timestamp = ls_timestamp.

      MOVE-CORRESPONDING ls_timestamp TO ls_save_h.

      APPEND ls_save_h TO lt_save_h.
      CLEAR  ls_save_h.

      " 오더전환여부 체크 - plan테이블에 ㅇㅇ
      MOVE-CORRESPONDING gs_plan TO ls_plan.
      ls_plan-versp = 'X'.
      APPEND ls_plan TO lt_plan.
      CLEAR : ls_plan, gs_plan.

    ENDIF.

  ENDLOOP.

  IF lt_save_h IS NOT INITIAL.
    MODIFY zte2pp0009 FROM TABLE lt_save_h.
    IF sy-subrc <> 0.
      lv_commit = 'X'.
    ENDIF.

    MODIFY zte2pp0007 FROM TABLE lt_plan.
    IF sy-subrc <> 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit <> 'X'.
    COMMIT WORK AND WAIT.
    gv_ok = 'X'.
    MESSAGE s000 WITH '생산오더로 전환이 완료되었습니다.'.
    PERFORM get_plan.
    PERFORM set_icon.
    CLEAR  : gs_info_bom, gs_info_rou.
    PERFORM refresh_grid.
    PERFORM refresh_screen.
    " 나중에 생산오더에없는거만 전환해야함.
    " 애초에 전환된거 versp = 'X'는 전환 못하게 !!!
  ELSE.
    ROLLBACK WORK.
    MESSAGE s000 WITH '생산오더 전환중 문제가 생겼습니다.' DISPLAY LIKE 'E'.

    CLEAR lv_commit.
  ENDIF.

ENDFORM.

FORM get_bom USING pv_matnr TYPE zte2mm0001-matnr.

  SELECT SINGLE stlnr bmeng meins
    INTO CORRESPONDING FIELDS OF gs_info_bom
    FROM zte2pp0002
   WHERE matnr = pv_matnr.

  SELECT a~idnrk a~menge a~meins
         b~maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_bom
    FROM zte2pp0002 AS c
   INNER JOIN zte2pp0003 AS a
      ON a~stlnr = c~stlnr
   INNER JOIN zte2mm0001 AS b
      ON a~idnrk = b~matnr
   WHERE c~matnr = pv_matnr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '해당 생산의 BOM이 존재하지 않습니다.' DISPLAY LIKE 'E'.

  ENDIF.

*&---------------------------------------------------------------------*
FORM get_rou USING pv_matnr TYPE zte2mm0001-matnr.

  SELECT SINGLE a~plnnr b~ktext c~werks b~runyn a~bmsch a~meins
    INTO CORRESPONDING FIELDS OF gs_info_rou
    FROM zte2pp0005 AS  a
   INNER JOIN zte2pp0004 AS b
      ON a~plnnr = b~plnnr
   INNER JOIN zte2pp0001 AS c
      ON a~arbpl = c~arbpl
   INNER JOIN zte2pp0008 AS d
      ON d~matnr = pv_matnr
     AND d~plnnr = b~plnnr.

  CASE gs_info_rou-runyn.
    WHEN 'Y'.
      gs_info_rou-text = '가동'.
    WHEN 'N'.
      gs_info_rou-text = '정지'.
  ENDCASE.


  SELECT a~vornr a~arbpl a~ltxa1
         b~ktext
    INTO CORRESPONDING FIELDS OF TABLE gt_rou
    FROM zte2pp0005 AS a
   INNER JOIN zte2pp0001 AS b
      ON a~arbpl = b~arbpl
   INNER JOIN zte2pp0008 AS c
      ON c~matnr = pv_matnr
     AND c~plnnr = a~plnnr
   ORDER BY a~vornr.

  IF sy-subrc <> 0 .
    MESSAGE s000 WITH '해당 생산의 라우팅이 존재하지 않습니다.' DISPLAY LIKE 'E'.

  ENDIF.
