FORM get_bom_h .

  DATA : lr_maktx TYPE RANGE OF zte2mm0001-maktx,
         ls_maktx LIKE LINE OF lr_maktx.

  CLEAR : gs_bom_h, ls_maktx.
  REFRESH : gt_bom_h, lr_maktx.

  IF pa_maktx IS NOT INITIAL.

    ls_maktx-sign = 'I'.
    ls_maktx-low = '*' && pa_maktx && '*'.
    ls_maktx-option = 'CP'.

    APPEND ls_maktx TO lr_maktx.

  ENDIF.

  SELECT a~stlnr, a~matnr, b~maktx, a~werks, a~bmeng, a~meins,
         CASE a~status
         WHEN 'A' THEN '활성'
         WHEN 'I' THEN '비활성'
         END AS stat_txt,
         a~status
    FROM zte2pp0002 AS a
   INNER JOIN zte2mm0001 AS b
      ON a~matnr = b~matnr
   WHERE a~matnr IN @so_matnr
     AND a~werks = @pa_werks
     AND b~maktx IN @lr_maktx
     AND a~stlnr IN @so_stlnr
   ORDER BY a~stlnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_bom_h.


ENDFORM.

  CASE pv_ucomm.
    WHEN 'APPEND'.
      PERFORM append_bom_i.
      PERFORM refresh_grid_i.

    WHEN 'DELETE'.
      PERFORM delete_bom_i.

      PERFORM refresh_grid_i.

    WHEN 'EDIT'.
      PERFORM edit_mode.

      PERFORM refresh_grid_h.

  ENDCASE.

ENDFORM.

*&      --> ET_GOOD_CELLS

FORM handle_changed_finished USING pv_modified
                                       pt_modi TYPE lvc_t_modi.

  DATA : ls_modi  TYPE lvc_s_modi,
         lv_maktx TYPE zte2mm0001-maktx.

  CHECK pv_modified IS NOT INITIAL.

  LOOP AT pt_modi INTO ls_modi.

    READ TABLE gt_bom_i INTO gs_bom_i INDEX ls_modi-row_id.

    IF sy-subrc <> 0.
      CONTINUE.
    ENDIF.

    gs_bom_i-mark = 'X'.

    CASE ls_modi-fieldname. " 나중에 단위필드도 막기.
      WHEN 'IDNRK'.

        SELECT SINGLE maktx
          INTO gs_bom_i-maktx
          FROM zte2mm0001
         WHERE matnr = gs_bom_i-idnrk.

    ENDCASE.

    MODIFY gt_bom_i FROM gs_bom_i INDEX ls_modi-row_id
    TRANSPORTING mark maktx.

  ENDLOOP.

  PERFORM refresh_grid_i.

ENDFORM.
